# 2-3 trees: the in-order entry list (abstraction) and completeness (every
# leaf at depth h, the balance invariant). Insertion refines the spec's
# sorted insertion and keeps completeness (possibly growing by one level).

def io(-K: Data, -V: Data, t: B.Tree<K, V>) -> List<&2, E.Entry<K, V>>:
  match t:
    case B.Leaf{}:
      Nil{}
    case B.Node2{l, e, r}:
      SC.append(E.Entry<K, V>, io(K, V, l), Con{e, io(K, V, r)})
    case B.Node3{l, e1, m, e2, r}:
      SC.append(E.Entry<K, V>, io(K, V, l), Con{e1, SC.append(E.Entry<K, V>, io(K, V, m), Con{e2, io(K, V, r)})})

def ioI(-K: Data, -V: Data, u: B.UpI<K, V>) -> List<&2, E.Entry<K, V>>:
  match u:
    case B.TI{t}:
      io(K, V, t)
    case B.OF{l, e, r}:
      SC.append(E.Entry<K, V>, io(K, V, l), Con{e, io(K, V, r)})

def cpl(-K: Data, -V: Data, h: Nat, t: B.Tree<K, V>) -> Bool:
  match h t:
    case 0n B.Leaf{}:
      True{}
    case 0n B.Node2{l, e, r}:
      False{}
    case 0n B.Node3{l, e1, m, e2, r}:
      False{}
    case 1n+p B.Leaf{}:
      False{}
    case 1n+ +p B.Node2{l, e, r}:
      Bool.and(cpl(K, V, p, l), cpl(K, V, p, r))
    case 1n+ +p B.Node3{l, e1, m, e2, r}:
      Bool.and(cpl(K, V, p, l), Bool.and(cpl(K, V, p, m), cpl(K, V, p, r)))

def cplI(-K: Data, -V: Data, +h: Nat, u: B.UpI<K, V>) -> Bool:
  match u:
    case B.TI{t}:
      cpl(K, V, h, t)
    case B.OF{l, e, r}:
      Bool.and(cpl(K, V, h, l), cpl(K, V, h, r))

def hI(-K: Data, -V: Data, +h: Nat, u: B.UpI<K, V>) -> Nat:
  match u:
    case B.TI{t}:
      h
    case B.OF{l, e, r}:
      1n+h

# Height along the leftmost path.
def ht(-K: Data, -V: Data, t: B.Tree<K, V>) -> Nat:
  match t:
    case B.Leaf{}:
      0n
    case B.Node2{l, e, r}:
      1n+ht(K, V, l)
    case B.Node3{l, e1, m, e2, r}:
      1n+ht(K, V, l)

def ht_cpl(-K: Data, -V: Data, +h: Nat, +t: B.Tree<K, V>, +pf: {cpl(K, V, h, t) == True{} : Bool}) -> {ht(K, V, t) == h : Nat}:
  match h t:
    case 0n B.Leaf{}:
      {==}
    case 0n B.Node2{l, e, r}:
      Empty.absurd({1n+ht(K, V, l) == 0n : Nat}, L.false_true(pf))
    case 0n B.Node3{l, e1, m, e2, r}:
      Empty.absurd({1n+ht(K, V, l) == 0n : Nat}, L.false_true(pf))
    case 1n+p B.Leaf{}:
      Empty.absurd({0n == 1n+p : Nat}, L.false_true(pf))
    case 1n+ +p B.Node2{+l, +e, +r}:
      N.succ_cong(ht(K, V, l), p, ht_cpl(K, V, p, l, L.and_left(cpl(K, V, p, l), cpl(K, V, p, r), pf)))
    case 1n+ +p B.Node3{+l, +e1, +m, +e2, +r}:
      N.succ_cong(ht(K, V, l), p, ht_cpl(K, V, p, l, L.and_left(cpl(K, V, p, l), Bool.and(cpl(K, V, p, m), cpl(K, V, p, r)), pf)))

# ---- sortedness facts at a node ----

def s2_l(~K: Data, ~cmp: K -> K -> Cmp, -V: Data, +l: List<&2, E.Entry<K, V>>, +e: E.Entry<K, V>, +r: List<&2, E.Entry<K, V>>, +s: {SO.sorted(~K, ~cmp, V, SC.append(E.Entry<K, V>, l, Con{e, r})) == True{} : Bool}) -> {SO.sorted(~K, ~cmp, V, l) == True{} : Bool}:
  SO.so_left(~K, ~cmp, V, l, Con{e, r}, s)

def s2_r(~K: Data, ~cmp: K -> K -> Cmp, -V: Data, +l: List<&2, E.Entry<K, V>>, +e: E.Entry<K, V>, +r: List<&2, E.Entry<K, V>>, +s: {SO.sorted(~K, ~cmp, V, SC.append(E.Entry<K, V>, l, Con{e, r})) == True{} : Bool}) -> {SO.sorted(~K, ~cmp, V, r) == True{} : Bool}:
  SO.so_tail(~K, ~cmp, V, e, r, SO.so_right(~K, ~cmp, V, l, Con{e, r}, s))

def s2_ab(~K: Data, ~cmp: K -> K -> Cmp, -V: Data, +l: List<&2, E.Entry<K, V>>, +e: E.Entry<K, V>, +r: List<&2, E.Entry<K, V>>, +s: {SO.sorted(~K, ~cmp, V, SC.append(E.Entry<K, V>, l, Con{e, r})) == True{} : Bool}) -> {SO.above(~K, ~cmp, V, r, E.key(K, V, e)) == True{} : Bool}:
  SO.so_above(~K, ~cmp, V, e, r, SO.so_right(~K, ~cmp, V, l, Con{e, r}, s))

def eq_flip(~K: Data, ~cmp: K -> K -> Cmp, ~o: O.Order(~K, ~cmp), +a: K, +b: K, +h: {cmp(a, b) == EQ{} : Cmp}) -> {cmp(b, a) == EQ{} : Cmp}:
  %Equal.sym(Cmp, cmp(b, a), O.flipc(cmp(a, b)), O.flip(~K, ~cmp, o, a, b)) : {_ == EQ{} : Cmp}
  %Equal.sym(Cmp, cmp(a, b), EQ{}, h) : {O.flipc(_) == EQ{} : Cmp}
  {==}

# Keys of the left part are below k when k equals / exceeds the node key.
def bl_eq(~K: Data, ~cmp: K -> K -> Cmp, ~o: O.Order(~K, ~cmp), -V: Data, +l: List<&2, E.Entry<K, V>>, +e: E.Entry<K, V>, +k: K, +hb: {SO.below(~K, ~cmp, V, l, E.key(K, V, e)) == True{} : Bool}, +h: {cmp(k, E.key(K, V, e)) == EQ{} : Cmp}) -> {SO.below(~K, ~cmp, V, l, k) == True{} : Bool}:
  SO.below_eq(~K, ~cmp, ~o, V, l, E.key(K, V, e), k, hb, eq_flip(~K, ~cmp, ~o, k, E.key(K, V, e), h))

def bl_gt(~K: Data, ~cmp: K -> K -> Cmp, ~o: O.Order(~K, ~cmp), -V: Data, +l: List<&2, E.Entry<K, V>>, +e: E.Entry<K, V>, +k: K, +hb: {SO.below(~K, ~cmp, V, l, E.key(K, V, e)) == True{} : Bool}, +h: {cmp(k, E.key(K, V, e)) == GT{} : Cmp}) -> {SO.below(~K, ~cmp, V, l, k) == True{} : Bool}:
  SO.below_lt(~K, ~cmp, ~o, V, l, E.key(K, V, e), k, hb, SO.lt_of_gt(~K, ~cmp, ~o, k, E.key(K, V, e), h))

def cplU(-K: Data, -V: Data, h: Nat, t: B.Tree<K, V>) -> Bool:
  match h:
    case 0n:
      False{}
    case 1n+p:
      cpl(K, V, p, t)

def cplD(-K: Data, -V: Data, +h: Nat, u: B.UpD<K, V>) -> Bool:
  match u:
    case B.TD{t}:
      cpl(K, V, h, t)
    case B.UF{t}:
      cplU(K, V, h, t)

def smC(-K: Data, -V: Data, +h: Nat, s: B.SM<K, V>) -> Bool:
  match s:
    case B.SMNone{}:
      True{}
    case B.SMSome{x, u}:
      cplD(K, V, h, u)

# ---- balance of the resulting root ----

def bal(-K: Data, -V: Data, +t: B.Tree<K, V>) -> Bool:
  cpl(K, V, ht(K, V, t), t)

def bal_of(-K: Data, -V: Data, +h: Nat, +t: B.Tree<K, V>, +c: {cpl(K, V, h, t) == True{} : Bool}) -> {bal(K, V, t) == True{} : Bool}:
  %Equal.sym(Nat, ht(K, V, t), h, ht_cpl(K, V, h, t, c)) : {cpl(K, V, _, t) == True{} : Bool}
  c

def topI_bal(-K: Data, -V: Data, +h: Nat, +u: B.UpI<K, V>, +cu: {cplI(K, V, h, u) == True{} : Bool}) -> {bal(K, V, B.treeI(K, V, u)) == True{} : Bool}:
  bal_of(K, V, hI(K, V, h, u), B.treeI(K, V, u), top_cpl(K, V, h, u, cu))

def topD_bal(-K: Data, -V: Data, +h: Nat, +u: B.UpD<K, V>, +cu: {cplD(K, V, h, u) == True{} : Bool}) -> {bal(K, V, B.treeD(K, V, u)) == True{} : Bool}:
  match h u:
    case _ B.TD{+t}:
      bal_of(K, V, h, t, cu)
    case 0n B.UF{t}:
      Empty.absurd({bal(K, V, t) == True{} : Bool}, L.false_true(cu))
    case 1n+ +p B.UF{+t}:
      bal_of(K, V, p, t, cu)

# Tree-level insert/remove preserve balance and refine the sorted list.
def insert_bal(~K: Data, ~cmp: K -> K -> Cmp, -V: Data, +t: B.Tree<K, V>, +k: K, +v: V, +b: {bal(K, V, t) == True{} : Bool}) -> {bal(K, V, B.insert_tree(~K, ~cmp, V, t, k, v)) == True{} : Bool}:
  topI_bal(K, V, ht(K, V, t), B.ins(~K, ~cmp, V, t, k, v, B.c1(~K, ~cmp, V, k, t), B.c2(~K, ~cmp, V, k, t)), ins_cpl(~K, ~cmp, V, ht(K, V, t), t, k, v, B.c1(~K, ~cmp, V, k, t), B.c2(~K, ~cmp, V, k, t), b))

def remove_bal(~K: Data, ~cmp: K -> K -> Cmp, -V: Data, +t: B.Tree<K, V>, +k: K, +b: {bal(K, V, t) == True{} : Bool}) -> {bal(K, V, B.remove_tree(~K, ~cmp, V, t, k)) == True{} : Bool}:
  topD_bal(K, V, ht(K, V, t), B.del(~K, ~cmp, V, t, k, B.c1(~K, ~cmp, V, k, t), B.c2(~K, ~cmp, V, k, t)), del_cpl(~K, ~cmp, V, ht(K, V, t), t, k, B.c1(~K, ~cmp, V, k, t), B.c2(~K, ~cmp, V, k, t), b))

def insert_io(~K: Data, ~cmp: K -> K -> Cmp, ~o: O.Order(~K, ~cmp), -V: Data, +t: B.Tree<K, V>, +k: K, +v: V, +s: {SO.sorted(~K, ~cmp, V, io(K, V, t)) == True{} : Bool}) -> {io(K, V, B.insert_tree(~K, ~cmp, V, t, k, v)) == S.ins(~K, ~cmp, V, k, v, io(K, V, t)) : List<&2, E.Entry<K, V>>}:

def remove_io(~K: Data, ~cmp: K -> K -> Cmp, ~o: O.Order(~K, ~cmp), -V: Data, +t: B.Tree<K, V>, +k: K, +s: {SO.sorted(~K, ~cmp, V, io(K, V, t)) == True{} : Bool}) -> {io(K, V, B.remove_tree(~K, ~cmp, V, t, k)) == S.del(~K, ~cmp, V, k, io(K, V, t)) : List<&2, E.Entry<K, V>>}:
