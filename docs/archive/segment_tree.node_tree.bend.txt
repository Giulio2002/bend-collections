import Base
import ./internal/vec.bend as V
import ../types/segment_tree.bend as E

# Segment tree over n U32 values with lazy range addition; all arithmetic
# wraps modulo 2^32.
#
# Representation: a perfect tree of depth d (2^d >= n). A leaf stores a
# value; an internal node stores a lazy tag z (pending addition for every
# value below it) and the sum of its subtree's values including its own tag
# (but excluding tags above it). The value at index i is the stored leaf
# plus all tags on the path to it.
#
#   get(i) / set(i, v)      walk the path, accumulating tags; set stores
#                           v minus the accumulated tags at the leaf
#   range_query(l, r)       prefix(r) - prefix(l); a prefix walk adds the
#                           sums of fully covered left subtrees
#   range_add(l, r, v)      lazy: add v to the first r values, then -v to the
#                           first l values; a fully covered subtree only
#                           receives a tag (O(1) node update), never a walk
#
# Errors never change the state: get/set with index >= n fail with
# IndexOutOfRange; range_query/range_add unless l <= r <= n fail with
# InvalidRange.
#
# Cost: every operation visits O(log n) nodes; each node visit scales a tag
# by the subtree width with U32.shln (O(log n) shifts), so get/set/
# range_query/range_add cost O(log^2 n) word operations. new O(log n)
# (shared zero subtrees); from_list O(n log^2 n); length O(1).

type Node is Data:
  SLeaf{value: U32}
  SNode{sum: U32, lazy: U32, lo: Node, hi: Node}

type SegTree is Data:
  ST{n: Nat, depth: Nat, root: Node}

def total(t: Node) -> U32:
  match t:
    case SLeaf{x}:
      x
    case SNode{s, z, l, r}:
      s

# Internal node over children of depth p with tag z.
def node(+p: Nat, +l: Node, +r: Node, +z: U32) -> Node:
  SNode{U32.add(U32.add(total(l), total(r)), U32.shln(z, 1n+p)), z, l, r}

def zeros(d: Nat) -> Node:
  match d:
    case 0n:
      SLeaf{0}
    case 1n+p:
      +k = zeros(p)
      SNode{0, 0, k, k}

# n zero values.
def new(+n: Nat) -> SegTree:
  +d = V.depth_for(n)
  ST{n, d, zeros(d)}

def length(t: SegTree) -> Nat:
  match t:
    case ST{n, d, r}:
      n

# Add v to every value of a depth-d subtree (lazily).
def tagall(d: Nat, t: Node, +v: U32) -> Node:
  match d t:
    case 0n SLeaf{x}:
      SLeaf{U32.add(x, v)}
    case 0n SNode{s, z, l, r}:
      SNode{s, z, l, r}
    case 1n+p SLeaf{x}:
      SLeaf{x}
    case 1n+p SNode{s, z, l, r}:
      SNode{U32.add(s, U32.shln(v, 1n+p)), U32.add(z, v), l, r}

def get_go(d: Nat, t: Node, +i: Nat, +a: U32, left: Bool) -> U32:
  match d t left:
    case 0n SLeaf{x} _:
      U32.add(x, a)
    case 0n SNode{s, z, l, r} _:
      0
    case 1n+p SLeaf{x} _:
      0
    case 1n+ +p SNode{s, +z, l, r} True{}:
      get_go(p, l, i, U32.add(a, z), V.dec(p, i))
    case 1n+ +p SNode{s, +z, l, r} False{}:
      get_go(p, r, Nat.sub(i, V.pow2(p)), U32.add(a, z), V.dec(p, Nat.sub(i, V.pow2(p))))

def set_go(d: Nat, t: Node, +i: Nat, +v: U32, +a: U32, left: Bool) -> Node:
  match d t left:
    case 0n SLeaf{x} _:
      SLeaf{U32.sub(v, a)}
    case 0n SNode{s, z, l, r} _:
      SNode{s, z, l, r}
    case 1n+p SLeaf{x} _:
      SLeaf{x}
    case 1n+ +p SNode{s, +z, l, r} True{}:
      node(p, set_go(p, l, i, v, U32.add(a, z), V.dec(p, i)), r, z)
    case 1n+ +p SNode{s, +z, l, r} False{}:
      node(p, l, set_go(p, r, Nat.sub(i, V.pow2(p)), v, U32.add(a, z), V.dec(p, Nat.sub(i, V.pow2(p)))), z)

def leaf_prefix(x: U32, i: Nat) -> U32:
  match i:
    case 0n:
      0
    case 1n+j:
      x

# Sum of the first i values of a depth-d subtree whose ancestors' tags add a.
def prefix_go(d: Nat, t: Node, +i: Nat, +a: U32, left: Bool) -> U32:
  match d t left:
    case 0n SLeaf{x} _:
      leaf_prefix(U32.add(x, a), i)
    case 0n SNode{s, z, l, r} _:
      0
    case 1n+p SLeaf{x} _:
      0
    case 1n+ +p SNode{s, +z, +l, r} True{}:
      prefix_go(p, l, i, U32.add(a, z), V.dec(p, i))
    case 1n+ +p SNode{s, +z, +l, r} False{}:
      U32.add(U32.add(total(l), U32.shln(U32.add(a, z), p)), prefix_go(p, r, Nat.sub(i, V.pow2(p)), U32.add(a, z), V.dec(p, Nat.sub(i, V.pow2(p)))))

def leaf_padd(+x: U32, i: Nat, +v: U32) -> Node:
  match i:
    case 0n:
      SLeaf{x}
    case 1n+j:
      SLeaf{U32.add(x, v)}

# Add v to the first i values of a depth-d subtree.
def padd_go(d: Nat, t: Node, +i: Nat, +v: U32, left: Bool) -> Node:
  match d t left:
    case 0n SLeaf{x} _:
      leaf_padd(x, i, v)
    case 0n SNode{s, z, l, r} _:
      SNode{s, z, l, r}
    case 1n+p SLeaf{x} _:
      SLeaf{x}
    case 1n+ +p SNode{s, +z, l, r} True{}:
      node(p, padd_go(p, l, i, v, V.dec(p, i)), r, z)
    case 1n+ +p SNode{s, +z, l, r} False{}:
      node(p, tagall(p, l, v), padd_go(p, r, Nat.sub(i, V.pow2(p)), v, V.dec(p, Nat.sub(i, V.pow2(p)))), z)

def prefix(+d: Nat, t: Node, +i: Nat) -> U32:
  prefix_go(d, t, i, 0, V.dec(d, i))

def padd(+d: Nat, t: Node, +i: Nat, +v: U32) -> Node:
  padd_go(d, t, i, v, V.dec(d, i))

def get_if(ok: Bool, +d: Nat, t: Node, +i: Nat) -> Result<&2, &2, E.Error, U32>:
  match ok:
    case True{}:
      Done{get_go(d, t, i, 0, V.dec(d, i))}
    case False{}:
      Fail{E.IndexOutOfRange{}}

# Value at index i.
def get(s: SegTree, +i: Nat) -> Result<&2, &2, E.Error, U32>:
  match s:
    case ST{+n, +d, t}:
      get_if(Nat.is_lt(i, n), d, t, i)

def set_if(ok: Bool, +n: Nat, +d: Nat, +t: Node, +i: Nat, +v: U32) -> SegTree & Result<&2, &2, E.Error, Unit>:
  match ok:
    case True{}:
      (ST{n, d, set_go(d, t, i, v, 0, V.dec(d, i))}, Done{Unit{}})
    case False{}:
      (ST{n, d, t}, Fail{E.IndexOutOfRange{}})

# Assign v at index i.
def set(s: SegTree, +i: Nat, +v: U32) -> SegTree & Result<&2, &2, E.Error, Unit>:
  match s:
    case ST{+n, +d, +t}:
      set_if(Nat.is_lt(i, n), n, d, t, i, v)

def query_if(ok: Bool, +d: Nat, +t: Node, +l: Nat, +r: Nat) -> Result<&2, &2, E.Error, U32>:
  match ok:
    case True{}:
      Done{U32.sub(prefix(d, t, r), prefix(d, t, l))}
    case False{}:
      Fail{E.InvalidRange{}}

# Sum of the values with index in [l, r).
def range_query(s: SegTree, +l: Nat, +r: Nat) -> Result<&2, &2, E.Error, U32>:
  match s:
    case ST{+n, +d, +t}:
      query_if(Bool.and(Nat.is_le(l, r), Nat.is_le(r, n)), d, t, l, r)

def add_if(ok: Bool, +n: Nat, +d: Nat, +t: Node, +l: Nat, +r: Nat, +v: U32) -> SegTree & Result<&2, &2, E.Error, Unit>:
  match ok:
    case True{}:
      (ST{n, d, padd(d, padd(d, t, r, v), l, U32.sub(0, v))}, Done{Unit{}})
    case False{}:
      (ST{n, d, t}, Fail{E.InvalidRange{}})

# Add v to every value with index in [l, r).
def range_add(s: SegTree, +l: Nat, +r: Nat, +v: U32) -> SegTree & Result<&2, &2, E.Error, Unit>:
  match s:
    case ST{+n, +d, +t}:
      add_if(Bool.and(Nat.is_le(l, r), Nat.is_le(r, n)), n, d, t, l, r, v)

def fill(xs: List<&2, U32>, +i: Nat, +s: SegTree) -> SegTree:
  match xs:
    case Nil{}:
      s
    case Con{x, rest}:
      fill(rest, 1n+i, Pair.fst(SegTree, Result<&2, &2, E.Error, Unit>, set(s, i, x)))

def count(xs: List<&2, U32>) -> Nat:
  match xs:
    case Nil{}:
      0n
    case Con{x, rest}:
      1n+count(rest)

# Segment tree holding exactly the given values.
def from_list(+xs: List<&2, U32>) -> SegTree:
  fill(xs, 0n, new(count(xs)))

# Trace runner.

def obs_unit(r: SegTree & Result<&2, &2, E.Error, Unit>) -> SegTree & E.Obs:
  (s, x) = r
  (s, E.OUnit{x})

def step(+s: SegTree, op: E.Op) -> SegTree & E.Obs:
  match op:
    case E.Length{}:
      (s, E.ONat{length(s)})
    case E.Get{i}:
      (s, E.OItem{get(s, i)})
    case E.Set{i, v}:
      obs_unit(set(s, i, v))
    case E.RangeQuery{l, r}:
      (s, E.OItem{range_query(s, l, r)})
    case E.RangeAdd{l, r, v}:
      obs_unit(range_add(s, l, r, v))

def record(acc: List<&2, E.Obs>, r: SegTree & E.Obs) -> SegTree & List<&2, E.Obs>:
  (s, o) = r
  (s, Con{o, acc})

def step_acc(op: E.Op, st: SegTree & List<&2, E.Obs>) -> SegTree & List<&2, E.Obs>:
  (s, acc) = st
  record(acc, step(s, op))

def run_acc(ops: List<&2, E.Op>, st: SegTree & List<&2, E.Obs>) -> SegTree & List<&2, E.Obs>:
  match ops:
    case Nil{}:
      st
    case Con{op, rest}:
      run_acc(rest, step_acc(op, st))

def finish(st: SegTree & List<&2, E.Obs>) -> SegTree & List<&2, E.Obs>:
  (s, acc) = st
  (s, List.reverse(&2, E.Obs, acc))

def run(ops: List<&2, E.Op>, s: SegTree) -> SegTree & List<&2, E.Obs>:
  finish(run_acc(ops, (s, Nil{})))
