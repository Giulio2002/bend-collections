# Abstraction: the sequence front ++ reverse(back).
def abs(-T: Data, d: DQ.Deque<T>) -> List<&2, T>:
  match d:
    case DQ.DQ{fl, f, bl, b}:
      SC.append(T, f, SC.reverse(T, b))

# Invariant: the cached lengths are exact.
def inv(-T: Data, d: DQ.Deque<T>) -> Bool:
  match d:
    case DQ.DQ{+fl, +f, +bl, +b}:
      Bool.and(Nat.is_eq(fl, SC.length(T, f)), Nat.is_eq(bl, SC.length(T, b)))

def is_nil(-T: Data, xs: List<&2, T>) -> Bool:
  match xs:
    case Nil{}:
      True{}
    case Con{h, t}:
      False{}

# After rebalancing for the front: an empty front means an empty deque.
def fok(-T: Data, d: DQ.Deque<T>) -> Bool:
  match d:
    case DQ.DQ{fl, Nil{}, bl, b}:
      is_nil(T, b)
    case DQ.DQ{fl, Con{h, t}, bl, b}:
      True{}

def bok(-T: Data, d: DQ.Deque<T>) -> Bool:
  match d:
    case DQ.DQ{fl, f, bl, Nil{}}:
      is_nil(T, f)
    case DQ.DQ{fl, f, bl, Con{h, t}}:
      True{}

def view_item(-T: Data, r: DQ.Deque<T> & Result<&2, &2, E.Error, T>) -> List<&2, T> & E.Obs<T>:
  match r:
    case Tuple{d, x}:
      (abs(T, d), E.OItem{x})

def nil_of(-T: Data, +xs: List<&2, T>, +h: {is_nil(T, xs) == True{} : Bool}) -> {xs == Nil{} : List<&2, T>}:
  match xs:
    case Nil{}:
      {==}
    case Con{x, t}:
      Empty.absurd({Con{x, t} == Nil{} : List<&2, T>}, L.false_true(h))

def inv_f(-T: Data, +fl: Nat, +f: List<&2, T>, +bl: Nat, +b: List<&2, T>, +g: {inv(T, DQ.DQ{fl, f, bl, b}) == True{} : Bool}) -> {fl == SC.length(T, f) : Nat}:
  N.eq_from_is_eq(fl, SC.length(T, f), L.and_left(Nat.is_eq(fl, SC.length(T, f)), Nat.is_eq(bl, SC.length(T, b)), g))

def inv_b(-T: Data, +fl: Nat, +f: List<&2, T>, +bl: Nat, +b: List<&2, T>, +g: {inv(T, DQ.DQ{fl, f, bl, b}) == True{} : Bool}) -> {bl == SC.length(T, b) : Nat}:
  N.eq_from_is_eq(bl, SC.length(T, b), L.and_right(Nat.is_eq(fl, SC.length(T, f)), Nat.is_eq(bl, SC.length(T, b)), g))

def inv_mk(-T: Data, +fl: Nat, +f: List<&2, T>, +bl: Nat, +b: List<&2, T>, +e1: {fl == SC.length(T, f) : Nat}, +e2: {bl == SC.length(T, b) : Nat}) -> {inv(T, DQ.DQ{fl, f, bl, b}) == True{} : Bool}:
  L.and_intro(Nat.is_eq(fl, SC.length(T, f)), Nat.is_eq(bl, SC.length(T, b)),
    L.subst(Nat, x => {Nat.is_eq(fl, x) == True{} : Bool}, fl, SC.length(T, f), e1, N.is_eq_refl(fl)),
    L.subst(Nat, x => {Nat.is_eq(bl, x) == True{} : Bool}, bl, SC.length(T, b), e2, N.is_eq_refl(bl)))

def half_le(+n: Nat) -> {Nat.is_le(DQ.half(n), n) == True{} : Bool}:
  match n:
    case 0n:
      {==}
    case 1n:
      {==}
    case 2n+q:
      N.le_trans(DQ.half(q), q, 1n+q, half_le(q), N.le_succ(q))

def half_succ_le(+m: Nat) -> {Nat.is_le(DQ.half(1n+m), m) == True{} : Bool}:
  match m:
    case 0n:
      {==}
    case 1n+q:
      half_le(q)

def sub1(+n: Nat) -> {Nat.sub(1n+n, 1n) == n : Nat}:
  N.sub_zero(n)

# A list of length 1 + m is a cons.
def len_cons_fok(-T: Data, +fl: Nat, +xs: List<&2, T>, +bl: Nat, +b: List<&2, T>, +m: Nat, +h: {SC.length(T, xs) == 1n+m : Nat}) -> {fok(T, DQ.DQ{fl, xs, bl, b}) == True{} : Bool}:
  match xs:
    case Nil{}:
      Empty.absurd({is_nil(T, b) == True{} : Bool}, N.zero_succ(m, h))
    case Con{x, t}:
      {==}

def len_cons_bok(-T: Data, +fl: Nat, +f: List<&2, T>, +bl: Nat, +xs: List<&2, T>, +m: Nat, +h: {SC.length(T, xs) == 1n+m : Nat}) -> {bok(T, DQ.DQ{fl, f, bl, xs}) == True{} : Bool}:
  match xs:
    case Nil{}:
      Empty.absurd({is_nil(T, f) == True{} : Bool}, N.zero_succ(m, h))
    case Con{x, t}:
      {==}

def len_brev(-T: Data, +xs: List<&2, T>) -> {SC.length(T, List.reverse(&2, T, xs)) == SC.length(T, xs) : Nat}:
  %Equal.sym(List<&2, T>, List.reverse(&2, T, xs), SC.reverse(T, xs), LL.base_rev(T, xs)) : {SC.length(T, _) == SC.length(T, xs) : Nat}
  LL.length_rev(T, xs)
