# Abstraction: the first n leaves. Invariant: perfect tree covering n with
# correct left-subtree sums.

def invf(+n: Nat, +d: Nat, +t: F.Node) -> Bool:
  Bool.and(T.perfect(d, t), Bool.and(Nat.is_le(n, V.pow2(d)), T.goodf(t)))

def inv(f: F.Fenwick) -> Bool:
  match f:
    case F.FT{n, d, t}:
      invf(n, d, t)

def abs(f: F.Fenwick) -> List<&2, U32>:
  match f:
    case F.FT{n, d, t}:
      SC.take(U32, T.leaves(t), n)

def view1(r: F.Fenwick & E.Obs) -> List<&2, U32> & E.Obs:
  match r:
    case Tuple{f, o}:
      (abs(f), o)

def StepOK(f: F.Fenwick, op: E.Op) -> Type:
  {view1(F.step(f, op)) == S.step(abs(f), op) : List<&2, U32> & E.Obs} & {inv(Pair.fst(F.Fenwick, E.Obs, F.step(f, op))) == True{} : Bool}

def i_pf(+n: Nat, +d: Nat, +t: F.Node, +g: {invf(n, d, t) == True{} : Bool}) -> {T.perfect(d, t) == True{} : Bool}:
  L.and_left(T.perfect(d, t), Bool.and(Nat.is_le(n, V.pow2(d)), T.goodf(t)), g)

def i_cap(+n: Nat, +d: Nat, +t: F.Node, +g: {invf(n, d, t) == True{} : Bool}) -> {Nat.is_le(n, V.pow2(d)) == True{} : Bool}:
  L.and_left(Nat.is_le(n, V.pow2(d)), T.goodf(t), L.and_right(T.perfect(d, t), Bool.and(Nat.is_le(n, V.pow2(d)), T.goodf(t)), g))

def i_good(+n: Nat, +d: Nat, +t: F.Node, +g: {invf(n, d, t) == True{} : Bool}) -> {T.goodf(t) == True{} : Bool}:
  L.and_right(Nat.is_le(n, V.pow2(d)), T.goodf(t), L.and_right(T.perfect(d, t), Bool.and(Nat.is_le(n, V.pow2(d)), T.goodf(t)), g))

def i_mk(+n: Nat, +d: Nat, +t: F.Node, +a: {T.perfect(d, t) == True{} : Bool}, +b: {Nat.is_le(n, V.pow2(d)) == True{} : Bool}, +c: {T.goodf(t) == True{} : Bool}) -> {invf(n, d, t) == True{} : Bool}:
  L.and_intro(T.perfect(d, t), Bool.and(Nat.is_le(n, V.pow2(d)), T.goodf(t)), a, L.and_intro(Nat.is_le(n, V.pow2(d)), T.goodf(t), b, c))

def n_le_len(+n: Nat, +d: Nat, +t: F.Node, +g: {invf(n, d, t) == True{} : Bool}) -> {Nat.is_le(n, SC.length(U32, T.leaves(t))) == True{} : Bool}:
  L.subst(Nat, q => {Nat.is_le(n, q) == True{} : Bool}, V.pow2(d), SC.length(U32, T.leaves(t)), Equal.sym(Nat, SC.length(U32, T.leaves(t)), V.pow2(d), T.leaves_length(d, t, i_pf(n, d, t, g))), i_cap(n, d, t, g))

def len_abs(+n: Nat, +d: Nat, +t: F.Node, +g: {invf(n, d, t) == True{} : Bool}) -> {SC.length(U32, SC.take(U32, T.leaves(t), n)) == n : Nat}:
  LL.sc_length_take(U32, T.leaves(t), n, n_le_len(n, d, t, g))

def spec_add_m(+xs: List<&2, U32>, +i: Nat, +v: U32, +hi: {Nat.is_lt(i, SC.length(U32, xs)) == True{} : Bool}, m: Maybe<&2, U32>, +em: {SC.nth(U32, xs, i) == m : Maybe<&2, U32>}) -> {S.add_at(m, xs, i, v) == (QS.addat(xs, i, v), E.OUnit{Done{Unit{}}}) : List<&2, U32> & E.Obs}:
  match m:
    case None{}:
      Empty.absurd({(xs, E.OUnit{Fail{E.IndexOutOfRange{}}}) == (QS.addat(xs, i, v), E.OUnit{Done{Unit{}}}) : List<&2, U32> & E.Obs}, LL.sc_nth_some(U32, xs, i, hi, em))
    case Some{y}:
      Equal.cong(List<&2, U32>, List<&2, U32> & E.Obs, zs => (zs, E.OUnit{Done{Unit{}}}), SC.update(U32, xs, i, U32.add(y, v)), QS.addat(xs, i, v), QS.update_addat(xs, i, y, v, em))

def AddOK(+n: Nat, +d: Nat, +t: F.Node, +i: Nat, +v: U32, b: Bool) -> Type:
  {view1(F.obs_unit(F.add_if(b, n, d, t, i, v))) == S.add_at(SC.nth(U32, SC.take(U32, T.leaves(t), n), i), SC.take(U32, T.leaves(t), n), i, v) : List<&2, U32> & E.Obs} & {inv(Pair.fst(F.Fenwick, E.Obs, F.obs_unit(F.add_if(b, n, d, t, i, v)))) == True{} : Bool}

def add_ok(+n: Nat, +d: Nat, +t: F.Node, +g: {invf(n, d, t) == True{} : Bool}, +i: Nat, +v: U32, b: Bool, +eb: {Nat.is_lt(i, n) == b : Bool}) -> AddOK(n, d, t, i, v, b):
  match b:
    case True{}:
      +A = SC.take(U32, T.leaves(t), n)
      +hi = N.lt_le_trans(i, n, V.pow2(d), eb, i_cap(n, d, t, g))
      +hA = L.subst(Nat, q => {Nat.is_lt(i, q) == True{} : Bool}, n, SC.length(U32, A), Equal.sym(Nat, SC.length(U32, A), n, len_abs(n, d, t, g)), eb)
      (%Equal.sym(List<&2, U32> & E.Obs, S.add_at(SC.nth(U32, A, i), A, i, v), (QS.addat(A, i, v), E.OUnit{Done{Unit{}}}), spec_add_m(A, i, v, hA, SC.nth(U32, A, i), {==})) : {(SC.take(U32, T.leaves(F.add_go(d, t, i, v, V.dec(d, i))), n), E.OUnit{Done{Unit{}}}) == _ : List<&2, U32> & E.Obs}
       %Equal.sym(List<&2, U32>, T.leaves(F.add_go(d, t, i, v, V.dec(d, i))), QS.addat(T.leaves(t), i, v), T.add_leaves(d, t, i, v, hi, V.dec(d, i), {==}, i_pf(n, d, t, g))) : {(SC.take(U32, _, n), E.OUnit{Done{Unit{}}}) == (QS.addat(A, i, v), E.OUnit{Done{Unit{}}}) : List<&2, U32> & E.Obs}
       %Equal.sym(List<&2, U32>, SC.take(U32, QS.addat(T.leaves(t), i, v), n), QS.addat(A, i, v), QS.take_addat(T.leaves(t), i, v, n)) : {(_, E.OUnit{Done{Unit{}}}) == (QS.addat(A, i, v), E.OUnit{Done{Unit{}}}) : List<&2, U32> & E.Obs}
       {==},
       i_mk(n, d, F.add_go(d, t, i, v, V.dec(d, i)), T.add_perfect(d, t, i, v, V.dec(d, i), i_pf(n, d, t, g)), i_cap(n, d, t, g), T.add_good(d, t, i, v, hi, V.dec(d, i), {==}, i_pf(n, d, t, g), i_good(n, d, t, g))))
    case False{}:
      +A = SC.take(U32, T.leaves(t), n)
      +hn = L.subst(Nat, q => {Nat.is_le(q, i) == True{} : Bool}, n, SC.length(U32, A), Equal.sym(Nat, SC.length(U32, A), n, len_abs(n, d, t, g)), N.not_lt_le(i, n, eb))
      (%Equal.sym(Maybe<&2, U32>, SC.nth(U32, A, i), None{}, LL.nth_none(U32, A, i, hn)) : {(A, E.OUnit{Fail{E.IndexOutOfRange{}}}) == S.add_at(_, A, i, v) : List<&2, U32> & E.Obs}
       {==}, g)

def prefix_eq(+n: Nat, +d: Nat, +t: F.Node, +g: {invf(n, d, t) == True{} : Bool}, +e: Nat, +he: {Nat.is_le(e, n) == True{} : Bool}) -> {F.prefix(d, t, e) == Q.sum(SC.take(U32, T.leaves(t), e)) : U32}:
  T.prefix_ok(d, t, e, N.le_trans(e, n, V.pow2(d), he, i_cap(n, d, t, g)), V.dec(d, e), {==}, i_pf(n, d, t, g), i_good(n, d, t, g))

def prefix_ok(+n: Nat, +d: Nat, +t: F.Node, +g: {invf(n, d, t) == True{} : Bool}, +e: Nat, b: Bool, +eb: {Nat.is_le(e, n) == b : Bool}) -> {F.prefix_if(b, d, t, e) == S.prefix_if(b, SC.take(U32, T.leaves(t), n), e) : Result<&2, &2, E.Error, U32>}:
  match b:
    case True{}:
      %Equal.sym(U32, F.prefix(d, t, e), Q.sum(SC.take(U32, T.leaves(t), e)), prefix_eq(n, d, t, g, e, eb)) : {Done{_} == Done{Q.sum(SC.take(U32, SC.take(U32, T.leaves(t), n), e))} : Result<&2, &2, E.Error, U32>}
      %Equal.sym(List<&2, U32>, SC.take(U32, SC.take(U32, T.leaves(t), n), e), SC.take(U32, T.leaves(t), e), LL.sc_take_take(U32, T.leaves(t), n, e, eb)) : {Done{Q.sum(SC.take(U32, T.leaves(t), e))} == Done{Q.sum(_)} : Result<&2, &2, E.Error, U32>}
      {==}
    case False{}:
      {==}

def range_ok(+n: Nat, +d: Nat, +t: F.Node, +g: {invf(n, d, t) == True{} : Bool}, +l: Nat, +r: Nat, b: Bool, +eb: {Bool.and(Nat.is_le(l, r), Nat.is_le(r, n)) == b : Bool}) -> {F.range_if(b, d, t, l, r) == S.range_if(b, SC.take(U32, T.leaves(t), n), l, r) : Result<&2, &2, E.Error, U32>}:
  match b:
    case True{}:
      +hlr = L.and_left(Nat.is_le(l, r), Nat.is_le(r, n), eb)
      +hrn = L.and_right(Nat.is_le(l, r), Nat.is_le(r, n), eb)
      %Equal.sym(U32, F.prefix(d, t, r), Q.sum(SC.take(U32, T.leaves(t), r)), prefix_eq(n, d, t, g, r, hrn)) : {Done{U32.sub(_, F.prefix(d, t, l))} == Done{Q.sum(Q.slice(SC.take(U32, T.leaves(t), n), l, r))} : Result<&2, &2, E.Error, U32>}
      %Equal.sym(U32, F.prefix(d, t, l), Q.sum(SC.take(U32, T.leaves(t), l)), prefix_eq(n, d, t, g, l, N.le_trans(l, r, n, hlr, hrn))) : {Done{U32.sub(Q.sum(SC.take(U32, T.leaves(t), r)), _)} == Done{Q.sum(Q.slice(SC.take(U32, T.leaves(t), n), l, r))} : Result<&2, &2, E.Error, U32>}
      %Equal.sym(U32, U32.sub(Q.sum(SC.take(U32, T.leaves(t), r)), Q.sum(SC.take(U32, T.leaves(t), l))), Q.sum(Q.slice(T.leaves(t), l, r)), QS.sum_slice(T.leaves(t), l, r, hlr)) : {Done{_} == Done{Q.sum(Q.slice(SC.take(U32, T.leaves(t), n), l, r))} : Result<&2, &2, E.Error, U32>}
      %Equal.sym(List<&2, U32>, Q.slice(SC.take(U32, T.leaves(t), n), l, r), Q.slice(T.leaves(t), l, r), QS.slice_take(T.leaves(t), n, l, r, hrn, hlr)) : {Done{Q.sum(Q.slice(T.leaves(t), l, r))} == Done{Q.sum(_)} : Result<&2, &2, E.Error, U32>}
      {==}
    case False{}:
      {==}

def step_ok(+f: F.Fenwick, +op: E.Op, +g: {inv(f) == True{} : Bool}) -> StepOK(f, op):
  match f op:
    case F.FT{+n, +d, +t} E.Length{}:
      (%Equal.sym(Nat, SC.length(U32, SC.take(U32, T.leaves(t), n)), n, len_abs(n, d, t, g)) : {(SC.take(U32, T.leaves(t), n), E.ONat{n}) == (SC.take(U32, T.leaves(t), n), E.ONat{_}) : List<&2, U32> & E.Obs}
       {==}, g)
    case F.FT{+n, +d, +t} E.Add{+i, +v}:
      add_ok(n, d, t, g, i, v, Nat.is_lt(i, n), {==})
    case F.FT{+n, +d, +t} E.PrefixSum{+e}:
      (%Equal.sym(Nat, SC.length(U32, SC.take(U32, T.leaves(t), n)), n, len_abs(n, d, t, g)) : {(SC.take(U32, T.leaves(t), n), E.OSum{F.prefix_if(Nat.is_le(e, n), d, t, e)}) == (SC.take(U32, T.leaves(t), n), E.OSum{S.prefix_if(Nat.is_le(e, _), SC.take(U32, T.leaves(t), n), e)}) : List<&2, U32> & E.Obs}
       Equal.cong(Result<&2, &2, E.Error, U32>, List<&2, U32> & E.Obs, x => (SC.take(U32, T.leaves(t), n), E.OSum{x}), F.prefix_if(Nat.is_le(e, n), d, t, e), S.prefix_if(Nat.is_le(e, n), SC.take(U32, T.leaves(t), n), e), prefix_ok(n, d, t, g, e, Nat.is_le(e, n), {==})), g)
    case F.FT{+n, +d, +t} E.RangeSum{+l, +r}:
      (%Equal.sym(Nat, SC.length(U32, SC.take(U32, T.leaves(t), n)), n, len_abs(n, d, t, g)) : {(SC.take(U32, T.leaves(t), n), E.OSum{F.range_if(Bool.and(Nat.is_le(l, r), Nat.is_le(r, n)), d, t, l, r)}) == (SC.take(U32, T.leaves(t), n), E.OSum{S.range_if(Bool.and(Nat.is_le(l, r), Nat.is_le(r, _)), SC.take(U32, T.leaves(t), n), l, r)}) : List<&2, U32> & E.Obs}
       Equal.cong(Result<&2, &2, E.Error, U32>, List<&2, U32> & E.Obs, x => (SC.take(U32, T.leaves(t), n), E.OSum{x}), F.range_if(Bool.and(Nat.is_le(l, r), Nat.is_le(r, n)), d, t, l, r), S.range_if(Bool.and(Nat.is_le(l, r), Nat.is_le(r, n)), SC.take(U32, T.leaves(t), n), l, r), range_ok(n, d, t, g, l, r, Bool.and(Nat.is_le(l, r), Nat.is_le(r, n)), {==})), g)

def new_inv(+n: Nat) -> {inv(F.new(n)) == True{} : Bool}:
  i_mk(n, V.depth_for(n), F.zeros(V.depth_for(n)), T.zeros_perfect(V.depth_for(n)), VL.depth_for(n), T.zeros_good(V.depth_for(n)))

def new_abs(+n: Nat) -> {abs(F.new(n)) == S.new(n) : List<&2, U32>}:
  %Equal.sym(List<&2, U32>, T.leaves(F.zeros(V.depth_for(n))), Q.zeros(V.pow2(V.depth_for(n))), T.zeros_leaves(V.depth_for(n))) : {SC.take(U32, _, n) == Q.zeros(n) : List<&2, U32>}
  LL.sc_take_rep(U32, V.pow2(V.depth_for(n)), 0, n, VL.depth_for(n))

def count_len(+xs: List<&2, U32>) -> {F.count(xs) == SC.length(U32, xs) : Nat}:
  match xs:
    case Nil{}:
      {==}
    case Con{x, t}:
      N.succ_cong(F.count(t), SC.length(U32, t), count_len(t))

def lt_add_succ(+a: Nat, +x: Nat) -> {Nat.is_lt(a, Nat.add(a, 1n+x)) == True{} : Bool}:
  %Equal.sym(Nat, Nat.add(a, 1n+x), 1n+Nat.add(a, x), N.add_succ(a, x)) : {Nat.is_lt(a, _) == True{} : Bool}
  N.le_lt_succ(a, Nat.add(a, x), N.le_add_right(a, x))

def PutOK(+n: Nat, +d: Nat, +t: F.Node, +p: List<&2, U32>, +x: U32, +r: List<&2, U32>) -> Type:
