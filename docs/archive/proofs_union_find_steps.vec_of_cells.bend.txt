# Abstraction, invariant and per-operation refinement for src/union_find.

def invf(+n: Nat, +d: Nat, +t: V.Vec<U.Cell>, +c: Nat) -> Bool:
  Bool.and(VL.perfect(U.Cell, d, t), Bool.and(Nat.is_le(n, V.pow2(d)), Bool.and(M.good(VL.slots(U.Cell, t), n, n), Nat.is_eq(c, M.cnt_fix(VL.slots(U.Cell, t), n)))))

def inv(s: U.UnionFind) -> Bool:
  match s:
    case U.UF{n, d, t, c}:
      invf(n, d, t, c)

def abs(s: U.UnionFind) -> List<&2, Nat>:
  match s:
    case U.UF{n, d, t, c}:
      M.labs(VL.slots(U.Cell, t), n)

def view1(r: U.UnionFind & E.Obs) -> List<&2, Nat> & E.Obs:
  match r:
    case Tuple{s, o}:
      (abs(s), o)

def StepOK(s: U.UnionFind, op: E.Op) -> Type:
  {view1(U.step(s, op)) == S.step(abs(s), op) : List<&2, Nat> & E.Obs} & {inv(Pair.fst(U.UnionFind, E.Obs, U.step(s, op))) == True{} : Bool}

# ---- invariant parts ----

def i_pf(+n: Nat, +d: Nat, +t: V.Vec<U.Cell>, +c: Nat, +g: {invf(n, d, t, c) == True{} : Bool}) -> {VL.perfect(U.Cell, d, t) == True{} : Bool}:
  L.and_left(VL.perfect(U.Cell, d, t), Bool.and(Nat.is_le(n, V.pow2(d)), Bool.and(M.good(VL.slots(U.Cell, t), n, n), Nat.is_eq(c, M.cnt_fix(VL.slots(U.Cell, t), n)))), g)

def i_r1(+n: Nat, +d: Nat, +t: V.Vec<U.Cell>, +c: Nat, +g: {invf(n, d, t, c) == True{} : Bool}) -> {Bool.and(Nat.is_le(n, V.pow2(d)), Bool.and(M.good(VL.slots(U.Cell, t), n, n), Nat.is_eq(c, M.cnt_fix(VL.slots(U.Cell, t), n)))) == True{} : Bool}:
  L.and_right(VL.perfect(U.Cell, d, t), Bool.and(Nat.is_le(n, V.pow2(d)), Bool.and(M.good(VL.slots(U.Cell, t), n, n), Nat.is_eq(c, M.cnt_fix(VL.slots(U.Cell, t), n)))), g)

def i_cap(+n: Nat, +d: Nat, +t: V.Vec<U.Cell>, +c: Nat, +g: {invf(n, d, t, c) == True{} : Bool}) -> {Nat.is_le(n, V.pow2(d)) == True{} : Bool}:
  L.and_left(Nat.is_le(n, V.pow2(d)), Bool.and(M.good(VL.slots(U.Cell, t), n, n), Nat.is_eq(c, M.cnt_fix(VL.slots(U.Cell, t), n))), i_r1(n, d, t, c, g))

def i_good(+n: Nat, +d: Nat, +t: V.Vec<U.Cell>, +c: Nat, +g: {invf(n, d, t, c) == True{} : Bool}) -> {M.good(VL.slots(U.Cell, t), n, n) == True{} : Bool}:
  L.and_left(M.good(VL.slots(U.Cell, t), n, n), Nat.is_eq(c, M.cnt_fix(VL.slots(U.Cell, t), n)), L.and_right(Nat.is_le(n, V.pow2(d)), Bool.and(M.good(VL.slots(U.Cell, t), n, n), Nat.is_eq(c, M.cnt_fix(VL.slots(U.Cell, t), n))), i_r1(n, d, t, c, g)))

def i_cnt(+n: Nat, +d: Nat, +t: V.Vec<U.Cell>, +c: Nat, +g: {invf(n, d, t, c) == True{} : Bool}) -> {c == M.cnt_fix(VL.slots(U.Cell, t), n) : Nat}:
  N.eq_from_is_eq(c, M.cnt_fix(VL.slots(U.Cell, t), n), L.and_right(M.good(VL.slots(U.Cell, t), n, n), Nat.is_eq(c, M.cnt_fix(VL.slots(U.Cell, t), n)), L.and_right(Nat.is_le(n, V.pow2(d)), Bool.and(M.good(VL.slots(U.Cell, t), n, n), Nat.is_eq(c, M.cnt_fix(VL.slots(U.Cell, t), n))), i_r1(n, d, t, c, g))))

def i_mk(+n: Nat, +d: Nat, +t: V.Vec<U.Cell>, +c: Nat, +a: {VL.perfect(U.Cell, d, t) == True{} : Bool}, +b: {Nat.is_le(n, V.pow2(d)) == True{} : Bool}, +e: {M.good(VL.slots(U.Cell, t), n, n) == True{} : Bool}, +f: {c == M.cnt_fix(VL.slots(U.Cell, t), n) : Nat}) -> {invf(n, d, t, c) == True{} : Bool}:
  L.and_intro(VL.perfect(U.Cell, d, t), Bool.and(Nat.is_le(n, V.pow2(d)), Bool.and(M.good(VL.slots(U.Cell, t), n, n), Nat.is_eq(c, M.cnt_fix(VL.slots(U.Cell, t), n)))), a,
    L.and_intro(Nat.is_le(n, V.pow2(d)), Bool.and(M.good(VL.slots(U.Cell, t), n, n), Nat.is_eq(c, M.cnt_fix(VL.slots(U.Cell, t), n))), b,
      L.and_intro(M.good(VL.slots(U.Cell, t), n, n), Nat.is_eq(c, M.cnt_fix(VL.slots(U.Cell, t), n)), e, M.eq_true(c, M.cnt_fix(VL.slots(U.Cell, t), n), f))))

def cap(+n: Nat, +d: Nat, +x: Nat, +hx: {Nat.is_lt(x, n) == True{} : Bool}, +hc: {Nat.is_le(n, V.pow2(d)) == True{} : Bool}) -> {Nat.is_lt(x, V.pow2(d)) == True{} : Bool}:
  N.lt_le_trans(x, n, V.pow2(d), hx, hc)

def len_slots(+n: Nat, +d: Nat, +t: V.Vec<U.Cell>, +c: Nat, +g: {invf(n, d, t, c) == True{} : Bool}) -> {Nat.is_le(n, SC.length(U.Cell, VL.slots(U.Cell, t))) == True{} : Bool}:
  L.subst(Nat, q => {Nat.is_le(n, q) == True{} : Bool}, V.pow2(d), SC.length(U.Cell, VL.slots(U.Cell, t)), Equal.sym(Nat, SC.length(U.Cell, VL.slots(U.Cell, t)), V.pow2(d), VL.slots_length(U.Cell, d, t, i_pf(n, d, t, c, g))), i_cap(n, d, t, c, g))

def cell_eq(+d: Nat, +t: V.Vec<U.Cell>, +x: Nat, +hx: {Nat.is_lt(x, V.pow2(d)) == True{} : Bool}, +pf: {VL.perfect(U.Cell, d, t) == True{} : Bool}) -> {U.cell(d, t, x) == M.lc(VL.slots(U.Cell, t), x) : U.Cell}:
  Equal.cong(Maybe<&2, U.Cell>, U.Cell, m => U.cell_of(m, x), V.get(U.Cell, d, t, x), SC.nth(U.Cell, VL.slots(U.Cell, t), x), VL.get(U.Cell, d, t, x, hx, pf))

def root_eq(+d: Nat, +t: V.Vec<U.Cell>, +x: Nat, +hx: {Nat.is_lt(x, V.pow2(d)) == True{} : Bool}, +pf: {VL.perfect(U.Cell, d, t) == True{} : Bool}) -> {U.root(d, t, x) == M.rt(VL.slots(U.Cell, t), x) : Nat}:
  Equal.cong(U.Cell, Nat, c => U.croot(c), U.cell(d, t, x), M.lc(VL.slots(U.Cell, t), x), cell_eq(d, t, x, hx, pf))

def relink_perfect(+ms: List<&2, Nat>, +d: Nat, +t: V.Vec<U.Cell>, +big: Nat, +pf: {VL.perfect(U.Cell, d, t) == True{} : Bool}) -> {VL.perfect(U.Cell, d, U.relink(ms, d, t, big)) == True{} : Bool}:
  match ms:
    case Nil{}:
      pf
    case Con{m, rest}:
      relink_perfect(rest, d, V.set(U.Cell, d, t, m, U.with_root(U.cell(d, t, m), big)), big, VL.set_perfect(U.Cell, d, t, m, U.with_root(U.cell(d, t, m), big), pf))

def relink_slots(+ms: List<&2, Nat>, +d: Nat, +t: V.Vec<U.Cell>, +big: Nat, +pf: {VL.perfect(U.Cell, d, t) == True{} : Bool}, +hb: {M.below(ms, V.pow2(d)) == True{} : Bool}) -> {VL.slots(U.Cell, U.relink(ms, d, t, big)) == UN.rl(ms, VL.slots(U.Cell, t), big) : List<&2, U.Cell>}:
  match ms:
    case Nil{}:
      {==}
    case Con{m, rest}:
      +hm = L.and_left(Nat.is_lt(m, V.pow2(d)), M.below(rest, V.pow2(d)), hb)
      +t1 = V.set(U.Cell, d, t, m, U.with_root(U.cell(d, t, m), big))
      +e1 = Equal.trans(List<&2, U.Cell>, VL.slots(U.Cell, t1), SC.update(U.Cell, VL.slots(U.Cell, t), m, U.with_root(U.cell(d, t, m), big)), SC.update(U.Cell, VL.slots(U.Cell, t), m, U.with_root(M.lc(VL.slots(U.Cell, t), m), big)),
        VL.set(U.Cell, d, t, m, U.with_root(U.cell(d, t, m), big), hm, pf),
        Equal.cong(U.Cell, List<&2, U.Cell>, c => SC.update(U.Cell, VL.slots(U.Cell, t), m, U.with_root(c, big)), U.cell(d, t, m), M.lc(VL.slots(U.Cell, t), m), cell_eq(d, t, m, hm, pf)))
      Equal.trans(List<&2, U.Cell>, VL.slots(U.Cell, U.relink(rest, d, t1, big)), UN.rl(rest, VL.slots(U.Cell, t1), big), UN.rl(rest, SC.update(U.Cell, VL.slots(U.Cell, t), m, U.with_root(M.lc(VL.slots(U.Cell, t), m), big)), big),
        relink_slots(rest, d, t1, big, VL.set_perfect(U.Cell, d, t, m, U.with_root(U.cell(d, t, m), big), pf), L.and_right(Nat.is_lt(m, V.pow2(d)), M.below(rest, V.pow2(d)), hb)),
        Equal.cong(List<&2, U.Cell>, List<&2, U.Cell>, x => UN.rl(rest, x, big), VL.slots(U.Cell, t1), SC.update(U.Cell, VL.slots(U.Cell, t), m, U.with_root(M.lc(VL.slots(U.Cell, t), m), big)), e1))

def below_mono(+m: List<&2, Nat>, +n: Nat, +k: Nat, +h: {M.below(m, n) == True{} : Bool}, +le: {Nat.is_le(n, k) == True{} : Bool}) -> {M.below(m, k) == True{} : Bool}:
  match m:
    case Nil{}:
      {==}
    case Con{x, t}:
      L.and_intro(Nat.is_lt(x, k), M.below(t, k), N.lt_le_trans(x, n, k, L.and_left(Nat.is_lt(x, n), M.below(t, n), h), le), below_mono(t, n, k, L.and_right(Nat.is_lt(x, n), M.below(t, n), h), le))

def merge_perfect(+d: Nat, +t: V.Vec<U.Cell>, +big: Nat, +small: Nat, +pf: {VL.perfect(U.Cell, d, t) == True{} : Bool}) -> {VL.perfect(U.Cell, d, U.merge(d, t, big, small)) == True{} : Bool}:
  VL.set_perfect(U.Cell, d, U.relink(U.cmembers(U.cell(d, t, small)), d, t, big), big, U.Cell{big, Nat.add(U.csize(U.cell(d, t, big)), U.csize(U.cell(d, t, small))), List.append(&2, Nat, U.cmembers(U.cell(d, t, small)), U.cmembers(U.cell(d, t, big)))}, relink_perfect(U.cmembers(U.cell(d, t, small)), d, t, big, pf))

def merge_slots(+n: Nat, +d: Nat, +t: V.Vec<U.Cell>, +c: Nat, +g: {invf(n, d, t, c) == True{} : Bool}, +big: Nat, +small: Nat, +h: {UN.H(VL.slots(U.Cell, t), n, big, small) == True{} : Bool}) -> {VL.slots(U.Cell, U.merge(d, t, big, small)) == UN.cl2(VL.slots(U.Cell, t), big, small) : List<&2, U.Cell>}:
  +cl = VL.slots(U.Cell, t)
  +pf = i_pf(n, d, t, c, g)
  +hc = i_cap(n, d, t, c, g)
  +hbig = cap(n, d, big, UN.h_big(cl, n, big, small, h), hc)
  +hsmall = cap(n, d, small, UN.h_small(cl, n, big, small, h), hc)
  +bel = below_mono(M.ms(cl, small), n, V.pow2(d), M.root_below(cl, n, small, UN.root_of_small(cl, n, big, small, h)), hc)
  %Equal.sym(U.Cell, U.cell(d, t, small), M.lc(cl, small), cell_eq(d, t, small, hsmall, pf)) : {VL.slots(U.Cell, V.set(U.Cell, d, U.relink(U.cmembers(_), d, t, big), big, U.Cell{big, Nat.add(U.csize(U.cell(d, t, big)), U.csize(_)), List.append(&2, Nat, U.cmembers(_), U.cmembers(U.cell(d, t, big)))})) == UN.cl2(cl, big, small) : List<&2, U.Cell>}
  %Equal.sym(U.Cell, U.cell(d, t, big), M.lc(cl, big), cell_eq(d, t, big, hbig, pf)) : {VL.slots(U.Cell, V.set(U.Cell, d, U.relink(M.ms(cl, small), d, t, big), big, U.Cell{big, Nat.add(U.csize(_), M.sz(cl, small)), List.append(&2, Nat, M.ms(cl, small), U.cmembers(_))})) == UN.cl2(cl, big, small) : List<&2, U.Cell>}
  %Equal.sym(List<&2, Nat>, List.append(&2, Nat, M.ms(cl, small), M.ms(cl, big)), SC.append(Nat, M.ms(cl, small), M.ms(cl, big)), LL.base_append(Nat, M.ms(cl, small), M.ms(cl, big))) : {VL.slots(U.Cell, V.set(U.Cell, d, U.relink(M.ms(cl, small), d, t, big), big, U.Cell{big, Nat.add(M.sz(cl, big), M.sz(cl, small)), _})) == UN.cl2(cl, big, small) : List<&2, U.Cell>}
  %Equal.sym(List<&2, U.Cell>, VL.slots(U.Cell, V.set(U.Cell, d, U.relink(M.ms(cl, small), d, t, big), big, UN.newcell(cl, big, small))), SC.update(U.Cell, VL.slots(U.Cell, U.relink(M.ms(cl, small), d, t, big)), big, UN.newcell(cl, big, small)), VL.set(U.Cell, d, U.relink(M.ms(cl, small), d, t, big), big, UN.newcell(cl, big, small), hbig, relink_perfect(M.ms(cl, small), d, t, big, pf))) : {_ == UN.cl2(cl, big, small) : List<&2, U.Cell>}
  %Equal.sym(List<&2, U.Cell>, VL.slots(U.Cell, U.relink(M.ms(cl, small), d, t, big)), UN.rl(M.ms(cl, small), cl, big), relink_slots(M.ms(cl, small), d, t, big, pf, bel)) : {SC.update(U.Cell, _, big, UN.newcell(cl, big, small)) == UN.cl2(cl, big, small) : List<&2, U.Cell>}
  {==}

def MergeOK(+n: Nat, +d: Nat, +t: V.Vec<U.Cell>, +c: Nat, +big: Nat, +small: Nat) -> Type:
  {abs(U.UF{n, d, U.merge(d, t, big, small), Nat.sub(c, 1n)}) == S.relabel(M.labs(VL.slots(U.Cell, t), n), small, big) : List<&2, Nat>} & {inv(U.UF{n, d, U.merge(d, t, big, small), Nat.sub(c, 1n)}) == True{} : Bool}

def merge_ok(+n: Nat, +d: Nat, +t: V.Vec<U.Cell>, +c: Nat, +g: {invf(n, d, t, c) == True{} : Bool}, +big: Nat, +small: Nat, +h: {UN.H(VL.slots(U.Cell, t), n, big, small) == True{} : Bool}) -> MergeOK(n, d, t, c, big, small):
  +cl = VL.slots(U.Cell, t)
  +es = Equal.sym(List<&2, U.Cell>, VL.slots(U.Cell, U.merge(d, t, big, small)), UN.cl2(cl, big, small), merge_slots(n, d, t, c, g, big, small, h))
  +ec = Equal.trans(Nat, Nat.sub(c, 1n), Nat.sub(1n+M.cnt_fix(UN.cl2(cl, big, small), n), 1n), M.cnt_fix(UN.cl2(cl, big, small), n),
    Equal.cong(Nat, Nat, q => Nat.sub(q, 1n), c, 1n+M.cnt_fix(UN.cl2(cl, big, small), n), Equal.trans(Nat, c, M.cnt_fix(cl, n), 1n+M.cnt_fix(UN.cl2(cl, big, small), n), i_cnt(n, d, t, c, g), UN.fix_count(cl, n, big, small, h))),
    N.sub_zero(M.cnt_fix(UN.cl2(cl, big, small), n)))
  (%es : {M.labs(_, n) == S.relabel(M.labs(cl, n), small, big) : List<&2, Nat>}
   UN.labs_relabel(cl, n, big, small, h, n, N.le_refl(n)),
   i_mk(n, d, U.merge(d, t, big, small), Nat.sub(c, 1n), merge_perfect(d, t, big, small, i_pf(n, d, t, c, g)), i_cap(n, d, t, c, g),
     %es : {M.good(_, n, n) == True{} : Bool}
     UN.good2(cl, n, big, small, h, n, N.le_refl(n)),
     %es : {Nat.sub(c, 1n) == M.cnt_fix(_, n) : Nat}
     ec))
