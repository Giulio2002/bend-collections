# Deque: public proof entry point.
#   abstraction  C.abs  (front ++ reverse(back))
#   invariant    C.inv  (cached lengths exact)
#   operations   step_ok (every operation; errors: pop/peek on empty)
#   traces       trace_new (arbitrary finite op lists from DQ.new)

def view1(-T: Data, r: DQ.Deque<T> & E.Obs<T>) -> List<&2, T> & E.Obs<T>:
  match r:
    case Tuple{d, o}:
      (C.abs(T, d), o)

def view(-T: Data, r: DQ.Deque<T> & List<&2, E.Obs<T>>) -> List<&2, T> & List<&2, E.Obs<T>>:
  match r:
    case Tuple{d, os}:
      (C.abs(T, d), os)

def StepOK(-T: Data, d: DQ.Deque<T>, op: E.Op<T>) -> Type:
  {view1(T, DQ.step(T, d, op)) == S.step(T, C.abs(T, d), op) : List<&2, T> & E.Obs<T>} & {C.inv(T, Pair.fst(DQ.Deque<T>, E.Obs<T>, DQ.step(T, d, op))) == True{} : Bool}

def vo(-T: Data, r: DQ.Deque<T> & Result<&2, &2, E.Error, T>) -> {view1(T, DQ.obs_item(T, r)) == C.view_item(T, r) : List<&2, T> & E.Obs<T>}:
  match r:
    case Tuple{d, x}:
      {==}

def fo(-T: Data, r: DQ.Deque<T> & Result<&2, &2, E.Error, T>) -> {Pair.fst(DQ.Deque<T>, E.Obs<T>, DQ.obs_item(T, r)) == Pair.fst(DQ.Deque<T>, Result<&2, &2, E.Error, T>, r) : DQ.Deque<T>}:
  match r:
    case Tuple{d, x}:
      {==}

def inv_item(-T: Data, r: DQ.Deque<T> & Result<&2, &2, E.Error, T>, +g: {C.inv(T, Pair.fst(DQ.Deque<T>, Result<&2, &2, E.Error, T>, r)) == True{} : Bool}) -> {C.inv(T, Pair.fst(DQ.Deque<T>, E.Obs<T>, DQ.obs_item(T, r))) == True{} : Bool}:
  L.subst(DQ.Deque<T>, x => {C.inv(T, x) == True{} : Bool}, Pair.fst(DQ.Deque<T>, Result<&2, &2, E.Error, T>, r), Pair.fst(DQ.Deque<T>, E.Obs<T>, DQ.obs_item(T, r)), Equal.sym(DQ.Deque<T>, Pair.fst(DQ.Deque<T>, E.Obs<T>, DQ.obs_item(T, r)), Pair.fst(DQ.Deque<T>, Result<&2, &2, E.Error, T>, r), fo(T, r)), g)

def step_ok(-T: Data, +d: DQ.Deque<T>, +op: E.Op<T>, +g: {C.inv(T, d) == True{} : Bool}) -> StepOK(T, d, op):
  match d op:
    case DQ.DQ{+fl, +f, +bl, +b} E.Length{}:
      +ef = C.inv_f(T, fl, f, bl, b, g)
      +eb = C.inv_b(T, fl, f, bl, b, g)
      (%Equal.sym(Nat, SC.length(T, SC.append(T, f, SC.reverse(T, b))), Nat.add(SC.length(T, f), SC.length(T, SC.reverse(T, b))), LL.length_append(T, f, SC.reverse(T, b))) : {(SC.append(T, f, SC.reverse(T, b)), E.ONat{Nat.add(fl, bl)}) == (SC.append(T, f, SC.reverse(T, b)), E.ONat{_}) : List<&2, T> & E.Obs<T>}
       %Equal.sym(Nat, SC.length(T, SC.reverse(T, b)), SC.length(T, b), LL.length_rev(T, b)) : {(SC.append(T, f, SC.reverse(T, b)), E.ONat{Nat.add(fl, bl)}) == (SC.append(T, f, SC.reverse(T, b)), E.ONat{Nat.add(SC.length(T, f), _)}) : List<&2, T> & E.Obs<T>}
       %ef : {(SC.append(T, f, SC.reverse(T, b)), E.ONat{Nat.add(fl, bl)}) == (SC.append(T, f, SC.reverse(T, b)), E.ONat{Nat.add(_, SC.length(T, b))}) : List<&2, T> & E.Obs<T>}
       %eb : {(SC.append(T, f, SC.reverse(T, b)), E.ONat{Nat.add(fl, bl)}) == (SC.append(T, f, SC.reverse(T, b)), E.ONat{Nat.add(fl, _)}) : List<&2, T> & E.Obs<T>}
       {==}, g)
    case DQ.DQ{+fl, +f, +bl, +b} E.PushFront{+x}:
      ({==}, C.inv_mk(T, 1n+fl, Con{x, f}, bl, b, N.succ_cong(fl, SC.length(T, f), C.inv_f(T, fl, f, bl, b, g)), C.inv_b(T, fl, f, bl, b, g)))
    case DQ.DQ{+fl, +f, +bl, +b} E.PushBack{+x}:
      (%Equal.sym(List<&2, T>, SC.append(T, f, SC.snoc(T, SC.reverse(T, b), x)), SC.snoc(T, SC.append(T, f, SC.reverse(T, b)), x), LL.append_snoc(T, f, SC.reverse(T, b), x)) : {(_, E.OUnit{}) == (SC.snoc(T, SC.append(T, f, SC.reverse(T, b)), x), E.OUnit{}) : List<&2, T> & E.Obs<T>}
       {==}, C.inv_mk(T, fl, f, 1n+bl, Con{x, b}, C.inv_f(T, fl, f, bl, b, g), N.succ_cong(bl, SC.length(T, b), C.inv_b(T, fl, f, bl, b, g))))
    case DQ.DQ{+fl, +f, +bl, +b} E.PopFront{}:
      +d0 = {DQ.DQ{fl, f, bl, b} : DQ.Deque<T>}
      +d1 = DQ.ready_front(T, d0)
      (%Equal.sym(List<&2, T> & E.Obs<T>, view1(T, DQ.obs_item(T, DQ.take_front(T, d1))), C.view_item(T, DQ.take_front(T, d1)), vo(T, DQ.take_front(T, d1))) : {_ == S.pop_front(T, C.abs(T, d0)) : List<&2, T> & E.Obs<T>}
       %Equal.sym(List<&2, T> & E.Obs<T>, C.view_item(T, DQ.take_front(T, d1)), S.pop_front(T, C.abs(T, d1)), EN.tf_view(T, d1, EN.rf_fok(T, fl, f, bl, b, g))) : {_ == S.pop_front(T, C.abs(T, d0)) : List<&2, T> & E.Obs<T>}
       %Equal.sym(List<&2, T>, C.abs(T, d1), C.abs(T, d0), EN.rf_abs(T, fl, f, bl, b)) : {S.pop_front(T, _) == S.pop_front(T, C.abs(T, d0)) : List<&2, T> & E.Obs<T>}
       {==},
       inv_item(T, DQ.take_front(T, d1), EN.tf_inv(T, d1, EN.rf_inv(T, fl, f, bl, b, g))))
    case DQ.DQ{+fl, +f, +bl, +b} E.PopBack{}:
      +d0 = {DQ.DQ{fl, f, bl, b} : DQ.Deque<T>}
      +d1 = DQ.ready_back(T, d0)
      (%Equal.sym(List<&2, T> & E.Obs<T>, view1(T, DQ.obs_item(T, DQ.take_back(T, d1))), C.view_item(T, DQ.take_back(T, d1)), vo(T, DQ.take_back(T, d1))) : {_ == S.pop_back(T, C.abs(T, d0)) : List<&2, T> & E.Obs<T>}

      (%Equal.sym(List<&2, T> & E.Obs<T>, view1(T, DQ.obs_item(T, DQ.peek_front_at(T, d1))), C.view_item(T, DQ.peek_front_at(T, d1)), vo(T, DQ.peek_front_at(T, d1))) : {_ == (C.abs(T, d0), E.OItem{S.item(T, SC.head(T, C.abs(T, d0)))}) : List<&2, T> & E.Obs<T>}
       %Equal.sym(List<&2, T> & E.Obs<T>, C.view_item(T, DQ.peek_front_at(T, d1)), (C.abs(T, d1), E.OItem{S.item(T, SC.head(T, C.abs(T, d1)))}), EN.pf_view(T, d1, EN.rf_fok(T, fl, f, bl, b, g))) : {_ == (C.abs(T, d0), E.OItem{S.item(T, SC.head(T, C.abs(T, d0)))}) : List<&2, T> & E.Obs<T>}
       %Equal.sym(List<&2, T>, C.abs(T, d1), C.abs(T, d0), EN.rf_abs(T, fl, f, bl, b)) : {(_, E.OItem{S.item(T, SC.head(T, _))}) == (C.abs(T, d0), E.OItem{S.item(T, SC.head(T, C.abs(T, d0)))}) : List<&2, T> & E.Obs<T>}

       inv_item(T, DQ.peek_front_at(T, d1), EN.pf_inv(T, d1, EN.rf_inv(T, fl, f, bl, b, g))))

      (%Equal.sym(List<&2, T> & E.Obs<T>, view1(T, DQ.obs_item(T, DQ.peek_back_at(T, d1))), C.view_item(T, DQ.peek_back_at(T, d1)), vo(T, DQ.peek_back_at(T, d1))) : {_ == (C.abs(T, d0), E.OItem{S.item(T, SC.last(T, C.abs(T, d0)))}) : List<&2, T> & E.Obs<T>}
       %Equal.sym(List<&2, T> & E.Obs<T>, C.view_item(T, DQ.peek_back_at(T, d1)), (C.abs(T, d1), E.OItem{S.item(T, SC.last(T, C.abs(T, d1)))}), EN.pb_view(T, d1, EN.rb_bok(T, fl, f, bl, b, g))) : {_ == (C.abs(T, d0), E.OItem{S.item(T, SC.last(T, C.abs(T, d0)))}) : List<&2, T> & E.Obs<T>}
       %Equal.sym(List<&2, T>, C.abs(T, d1), C.abs(T, d0), EN.rb_abs(T, fl, f, bl, b)) : {(_, E.OItem{S.item(T, SC.last(T, _))}) == (C.abs(T, d0), E.OItem{S.item(T, SC.last(T, C.abs(T, d0)))}) : List<&2, T> & E.Obs<T>}
       {==},

    case DQ.DQ{+fl, +f, +bl, +b} E.ToList{}:
      (%Equal.sym(List<&2, T>, List.reverse(&2, T, b), SC.reverse(T, b), LL.base_rev(T, b)) : {(SC.append(T, f, SC.reverse(T, b)), E.OList{List.append(&2, T, f, _)}) == (SC.append(T, f, SC.reverse(T, b)), E.OList{SC.append(T, f, SC.reverse(T, b))}) : List<&2, T> & E.Obs<T>}

# ---- arbitrary finite traces ----

def srun_obs(-T: Data, ops: List<&2, E.Op<T>>, xs: List<&2, T>) -> List<&2, E.Obs<T>>:
  Pair.snd(List<&2, T>, List<&2, E.Obs<T>>, S.run(T, ops, xs))

def srun_state(-T: Data, ops: List<&2, E.Op<T>>, xs: List<&2, T>) -> List<&2, T>:
  Pair.fst(List<&2, T>, List<&2, E.Obs<T>>, S.run(T, ops, xs))

def next(-T: Data, d: DQ.Deque<T>, op: E.Op<T>) -> DQ.Deque<T>:
  Pair.fst(DQ.Deque<T>, E.Obs<T>, DQ.step(T, d, op))

def obs(-T: Data, d: DQ.Deque<T>, op: E.Op<T>) -> E.Obs<T>:
  Pair.snd(DQ.Deque<T>, E.Obs<T>, DQ.step(T, d, op))

def run_cons(-T: Data, +op: E.Op<T>, +rest: List<&2, E.Op<T>>, +d: DQ.Deque<T>, +acc: List<&2, E.Obs<T>>) -> {DQ.run_acc(T, Con{op, rest}, (d, acc)) == DQ.run_acc(T, rest, (next(T, d, op), Con{obs(T, d, op), acc})) : DQ.Deque<T> & List<&2, E.Obs<T>>}:
  %Equal.sym(DQ.Deque<T> & E.Obs<T>, DQ.step(T, d, op), (next(T, d, op), obs(T, d, op)), L.pair_eta(DQ.Deque<T>, E.Obs<T>, DQ.step(T, d, op))) : {DQ.run_acc(T, rest, DQ.record(T, acc, _)) == DQ.run_acc(T, rest, (next(T, d, op), Con{obs(T, d, op), acc})) : DQ.Deque<T> & List<&2, E.Obs<T>>}
  {==}

def spec_view(-T: Data, +d: DQ.Deque<T>, +op: E.Op<T>, +g: {C.inv(T, d) == True{} : Bool}) -> {(C.abs(T, next(T, d, op)), obs(T, d, op)) == S.step(T, C.abs(T, d), op) : List<&2, T> & E.Obs<T>}:
  %Pair.fst({view1(T, DQ.step(T, d, op)) == S.step(T, C.abs(T, d), op) : List<&2, T> & E.Obs<T>}, {C.inv(T, next(T, d, op)) == True{} : Bool}, step_ok(T, d, op, g)) : {(C.abs(T, next(T, d, op)), obs(T, d, op)) == _ : List<&2, T> & E.Obs<T>}
  %Equal.sym(DQ.Deque<T> & E.Obs<T>, DQ.step(T, d, op), (next(T, d, op), obs(T, d, op)), L.pair_eta(DQ.Deque<T>, E.Obs<T>, DQ.step(T, d, op))) : {(C.abs(T, next(T, d, op)), obs(T, d, op)) == view1(T, _) : List<&2, T> & E.Obs<T>}
  {==}

def spec_run_cons(-T: Data, +op: E.Op<T>, +rest: List<&2, E.Op<T>>, +m: List<&2, T>, +m1: List<&2, T>, +o: E.Obs<T>, +e: {(m1, o) == S.step(T, m, op) : List<&2, T> & E.Obs<T>}) -> {S.run(T, Con{op, rest}, m) == (srun_state(T, rest, m1), Con{o, srun_obs(T, rest, m1)}) : List<&2, T> & List<&2, E.Obs<T>>}:
  %e : {S.cons_obs(T, Pair.snd(List<&2, T>, E.Obs<T>, _), S.run(T, rest, Pair.fst(List<&2, T>, E.Obs<T>, _))) == (srun_state(T, rest, m1), Con{o, srun_obs(T, rest, m1)}) : List<&2, T> & List<&2, E.Obs<T>>}
  %Equal.sym(List<&2, T> & List<&2, E.Obs<T>>, S.run(T, rest, m1), (srun_state(T, rest, m1), srun_obs(T, rest, m1)), L.pair_eta(List<&2, T>, List<&2, E.Obs<T>>, S.run(T, rest, m1))) : {S.cons_obs(T, o, _) == (srun_state(T, rest, m1), Con{o, srun_obs(T, rest, m1)}) : List<&2, T> & List<&2, E.Obs<T>>}
  {==}

def fin(-T: Data, ops: List<&2, E.Op<T>>, d: DQ.Deque<T>, acc: List<&2, E.Obs<T>>) -> DQ.Deque<T>:
  Pair.fst(DQ.Deque<T>, List<&2, E.Obs<T>>, DQ.run_acc(T, ops, (d, acc)))

def RunOK(-T: Data, ops: List<&2, E.Op<T>>, d: DQ.Deque<T>, acc: List<&2, E.Obs<T>>) -> Type:
  {DQ.run_acc(T, ops, (d, acc)) == (fin(T, ops, d, acc), List.reverse.go(&2, E.Obs<T>, srun_obs(T, ops, C.abs(T, d)), acc)) : DQ.Deque<T> & List<&2, E.Obs<T>>} & ({C.abs(T, fin(T, ops, d, acc)) == srun_state(T, ops, C.abs(T, d)) : List<&2, T>} & {C.inv(T, fin(T, ops, d, acc)) == True{} : Bool})

def run_ok(-T: Data, +ops: List<&2, E.Op<T>>, +d: DQ.Deque<T>, +acc: List<&2, E.Obs<T>>, +g: {C.inv(T, d) == True{} : Bool}) -> RunOK(T, ops, d, acc):
  match ops:
    case Nil{}:
      ({==}, ({==}, g))
    case Con{+op, +rest}:
      +d1 = next(T, d, op)
      +o = obs(T, d, op)
      +acc1 = {Con{o, acc} : List<&2, E.Obs<T>>}
      +g1 = Pair.snd({view1(T, DQ.step(T, d, op)) == S.step(T, C.abs(T, d), op) : List<&2, T> & E.Obs<T>}, {C.inv(T, d1) == True{} : Bool}, step_ok(T, d, op, g))
      +es = spec_run_cons(T, op, rest, C.abs(T, d), C.abs(T, d1), o, spec_view(T, d, op, g))
      +rc = run_cons(T, op, rest, d, acc)
      -P1 = {DQ.run_acc(T, rest, (d1, acc1)) == (fin(T, rest, d1, acc1), List.reverse.go(&2, E.Obs<T>, srun_obs(T, rest, C.abs(T, d1)), acc1)) : DQ.Deque<T> & List<&2, E.Obs<T>>}
      -P2 = {C.abs(T, fin(T, rest, d1, acc1)) == srun_state(T, rest, C.abs(T, d1)) : List<&2, T>}
      -P3 = {C.inv(T, fin(T, rest, d1, acc1)) == True{} : Bool}
      -I23 = P2 & P3
      (%Equal.sym(DQ.Deque<T> & List<&2, E.Obs<T>>, DQ.run_acc(T, Con{op, rest}, (d, acc)), DQ.run_acc(T, rest, (d1, acc1)), rc) : {_ == (Pair.fst(DQ.Deque<T>, List<&2, E.Obs<T>>, _), List.reverse.go(&2, E.Obs<T>, srun_obs(T, Con{op, rest}, C.abs(T, d)), acc)) : DQ.Deque<T> & List<&2, E.Obs<T>>}
       %Equal.sym(List<&2, T> & List<&2, E.Obs<T>>, S.run(T, Con{op, rest}, C.abs(T, d)), (srun_state(T, rest, C.abs(T, d1)), Con{o, srun_obs(T, rest, C.abs(T, d1))}), es) : {DQ.run_acc(T, rest, (d1, acc1)) == (fin(T, rest, d1, acc1), List.reverse.go(&2, E.Obs<T>, Pair.snd(List<&2, T>, List<&2, E.Obs<T>>, _), acc)) : DQ.Deque<T> & List<&2, E.Obs<T>>}
       Pair.fst(P1, I23, run_ok(T, rest, d1, acc1, g1)),
       (%Equal.sym(DQ.Deque<T> & List<&2, E.Obs<T>>, DQ.run_acc(T, Con{op, rest}, (d, acc)), DQ.run_acc(T, rest, (d1, acc1)), rc) : {C.abs(T, Pair.fst(DQ.Deque<T>, List<&2, E.Obs<T>>, _)) == srun_state(T, Con{op, rest}, C.abs(T, d)) : List<&2, T>}
        %Equal.sym(List<&2, T> & List<&2, E.Obs<T>>, S.run(T, Con{op, rest}, C.abs(T, d)), (srun_state(T, rest, C.abs(T, d1)), Con{o, srun_obs(T, rest, C.abs(T, d1))}), es) : {C.abs(T, fin(T, rest, d1, acc1)) == Pair.fst(List<&2, T>, List<&2, E.Obs<T>>, _) : List<&2, T>}
        Pair.fst(P2, P3, Pair.snd(P1, I23, run_ok(T, rest, d1, acc1, g1))),
        %Equal.sym(DQ.Deque<T> & List<&2, E.Obs<T>>, DQ.run_acc(T, Con{op, rest}, (d, acc)), DQ.run_acc(T, rest, (d1, acc1)), rc) : {C.inv(T, Pair.fst(DQ.Deque<T>, List<&2, E.Obs<T>>, _)) == True{} : Bool}
        Pair.snd(P2, P3, Pair.snd(P1, I23, run_ok(T, rest, d1, acc1, g1)))))

# Public trace law: any finite op list run from DQ.new through DQ.run equals
# the specification run on the empty sequence; the final deque satisfies inv.
def trace_new(-T: Data, +ops: List<&2, E.Op<T>>) -> {view(T, DQ.run(T, ops, DQ.new(T))) == S.run(T, ops, Nil{}) : List<&2, T> & List<&2, E.Obs<T>>}:
  +d0 = DQ.new(T)
  -P1 = {DQ.run_acc(T, ops, (d0, Nil{})) == (fin(T, ops, d0, Nil{}), List.reverse.go(&2, E.Obs<T>, srun_obs(T, ops, Nil{}), Nil{})) : DQ.Deque<T> & List<&2, E.Obs<T>>}
  -P2 = {C.abs(T, fin(T, ops, d0, Nil{})) == srun_state(T, ops, Nil{}) : List<&2, T>}
  -P3 = {C.inv(T, fin(T, ops, d0, Nil{})) == True{} : Bool}
  %Equal.sym(DQ.Deque<T> & List<&2, E.Obs<T>>, DQ.run_acc(T, ops, (d0, Nil{})), (fin(T, ops, d0, Nil{}), List.reverse.go(&2, E.Obs<T>, srun_obs(T, ops, Nil{}), Nil{})), Pair.fst(P1, P2 & P3, run_ok(T, ops, d0, Nil{}, {==}))) : {view(T, DQ.finish(T, _)) == S.run(T, ops, Nil{}) : List<&2, T> & List<&2, E.Obs<T>>}
