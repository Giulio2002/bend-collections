# Abstraction: the first n values (vals of the root). Invariant: perfect
# tree covering n whose nodes store the sums of their subtrees.

def invf(+n: Nat, +d: Nat, +t: T.Node) -> Bool:
  Bool.and(M.perfect(d, t), Bool.and(Nat.is_le(n, V.pow2(d)), M.goods(t)))

def inv(s: T.SegTree) -> Bool:
  match s:
    case T.ST{n, d, t}:
      invf(n, d, t)

def abs(s: T.SegTree) -> List<&2, U32>:
  match s:
    case T.ST{n, d, t}:
      SC.take(U32, M.vals(t), n)

def view1(r: T.SegTree & E.Obs) -> List<&2, U32> & E.Obs:
  match r:
    case Tuple{s, o}:
      (abs(s), o)

def StepOK(s: T.SegTree, op: E.Op) -> Type:
  {view1(T.step(s, op)) == S.step(abs(s), op) : List<&2, U32> & E.Obs} & {inv(Pair.fst(T.SegTree, E.Obs, T.step(s, op))) == True{} : Bool}

def i_pf(+n: Nat, +d: Nat, +t: T.Node, +g: {invf(n, d, t) == True{} : Bool}) -> {M.perfect(d, t) == True{} : Bool}:
  L.and_left(M.perfect(d, t), Bool.and(Nat.is_le(n, V.pow2(d)), M.goods(t)), g)

def i_cap(+n: Nat, +d: Nat, +t: T.Node, +g: {invf(n, d, t) == True{} : Bool}) -> {Nat.is_le(n, V.pow2(d)) == True{} : Bool}:
  L.and_left(Nat.is_le(n, V.pow2(d)), M.goods(t), L.and_right(M.perfect(d, t), Bool.and(Nat.is_le(n, V.pow2(d)), M.goods(t)), g))

def i_good(+n: Nat, +d: Nat, +t: T.Node, +g: {invf(n, d, t) == True{} : Bool}) -> {M.goods(t) == True{} : Bool}:
  L.and_right(Nat.is_le(n, V.pow2(d)), M.goods(t), L.and_right(M.perfect(d, t), Bool.and(Nat.is_le(n, V.pow2(d)), M.goods(t)), g))

def i_mk(+n: Nat, +d: Nat, +t: T.Node, +a: {M.perfect(d, t) == True{} : Bool}, +b: {Nat.is_le(n, V.pow2(d)) == True{} : Bool}, +c: {M.goods(t) == True{} : Bool}) -> {invf(n, d, t) == True{} : Bool}:
  L.and_intro(M.perfect(d, t), Bool.and(Nat.is_le(n, V.pow2(d)), M.goods(t)), a, L.and_intro(Nat.is_le(n, V.pow2(d)), M.goods(t), b, c))

def len_abs(+n: Nat, +d: Nat, +t: T.Node, +g: {invf(n, d, t) == True{} : Bool}) -> {SC.length(U32, SC.take(U32, M.vals(t), n)) == n : Nat}:
  LL.sc_length_take(U32, M.vals(t), n, L.subst(Nat, q => {Nat.is_le(n, q) == True{} : Bool}, V.pow2(d), SC.length(U32, M.vals(t)), Equal.sym(Nat, SC.length(U32, M.vals(t)), V.pow2(d), M.vals_length(d, t, i_pf(n, d, t, g))), i_cap(n, d, t, g)))

def vals0(+t: T.Node) -> {Q.add_each(M.vals(t), 0) == M.vals(t) : List<&2, U32>}:
  M.each_zero(M.vals(t))

       {==}, g)
    case T.ST{+n, +d, +t} E.Get{+i}:
      (Equal.cong(Result<&2, &2, E.Error, U32>, List<&2, U32> & E.Obs, x => (SC.take(U32, M.vals(t), n), E.OItem{x}), T.get_if(Nat.is_lt(i, n), d, t, i), S.item(SC.nth(U32, SC.take(U32, M.vals(t), n), i)), get_ok(n, d, t, g, i, Nat.is_lt(i, n), {==})), g)

      set_ok(n, d, t, g, i, v, Nat.is_lt(i, n), {==})
    case T.ST{+n, +d, +t} E.RangeQuery{+l, +r}:
      (%Equal.sym(Nat, SC.length(U32, SC.take(U32, M.vals(t), n)), n, len_abs(n, d, t, g)) : {(SC.take(U32, M.vals(t), n), E.OItem{T.query_if(Bool.and(Nat.is_le(l, r), Nat.is_le(r, n)), d, t, l, r)}) == (SC.take(U32, M.vals(t), n), E.OItem{S.query_if(Bool.and(Nat.is_le(l, r), Nat.is_le(r, _)), SC.take(U32, M.vals(t), n), l, r)}) : List<&2, U32> & E.Obs}

def lt_add_succ(+a: Nat, +x: Nat) -> {Nat.is_lt(a, Nat.add(a, 1n+x)) == True{} : Bool}:
  %Equal.sym(Nat, Nat.add(a, 1n+x), 1n+Nat.add(a, x), N.add_succ(a, x)) : {Nat.is_lt(a, _) == True{} : Bool}
  N.le_lt_succ(a, Nat.add(a, x), N.le_add_right(a, x))

def from_ok(+xs: List<&2, U32>) -> FillOK(xs, Nil{}, T.new(T.count(xs))):
  fill_ok(xs, Nil{}, T.new(T.count(xs)), new_inv(T.count(xs)),
    %count_len(xs) : {abs(T.new(T.count(xs))) == Q.zeros(_) : List<&2, U32>}
    new_abs(T.count(xs)))

def from_abs(+xs: List<&2, U32>) -> {abs(T.from_list(xs)) == S.from_list(xs) : List<&2, U32>}:
  Pair.fst({abs(T.from_list(xs)) == xs : List<&2, U32>}, {inv(T.from_list(xs)) == True{} : Bool}, from_ok(xs))

def from_inv(+xs: List<&2, U32>) -> {inv(T.from_list(xs)) == True{} : Bool}:
  Pair.snd({abs(T.from_list(xs)) == xs : List<&2, U32>}, {inv(T.from_list(xs)) == True{} : Bool}, from_ok(xs))
