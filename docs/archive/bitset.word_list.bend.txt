import Base
import ../types/bitset.bend as E

# Packed fixed-size bitset.
#
# A bitset of logical size `len` stores its bits LSB-first in a list of U32
# words: bit i lives in word i/32 at bit position i%32. The word list holds
# exactly ceil(len/32) words, and every bit at a position >= len (the unused
# tail of the last word) is zero. Both facts are the invariant proven in
# proofs/bitset.bend; union/intersection/difference/xor rely on it (e.g.
# difference is a AND NOT b, which stays masked because a's tail is zero).
#
# Errors never change the state: get/set/clear with index >= len fail with
# IndexOutOfRange; binary operations on operands of different logical size
# fail with LengthMismatch and return the left operand unchanged.
#
# Cost (words = ceil(len/32)): new O(words); length O(1); get/set/clear walk
# the word list, O(words); count/to_list O(32 * words); binary operations
# O(words). The word list is a Base List, not a random-access array.

type Bitset is Data:
  BS{len: Nat, words: List<&2, U32>}

# Word-level operations (native U32 arithmetic).

type WordOp is Data:
  KOr{}
  KAnd{}
  KDiff{}
  KXor{}

def low(w: U32) -> Bool:
  Bool.not(U32.is_even(w))

def word_get(w: U32, k: Nat) -> Bool:
  low(U32.shrn(w, k))

def word_put(v: Bool, w: U32, k: Nat) -> U32:
  match v:
    case True{}:
      U32.or(w, U32.shln(1, k))
    case False{}:
      U32.and(w, U32.not(U32.shln(1, k)))

def word_op(k: WordOp, a: U32, b: U32) -> U32:
  match k:
    case KOr{}:
      U32.or(a, b)
    case KAnd{}:
      U32.and(a, b)
    case KDiff{}:
      U32.and(a, U32.not(b))
    case KXor{}:
      U32.xor(a, b)

def bit_value(b: Bool) -> Nat:
  match b:
    case True{}:
      1n
    case False{}:
      0n

# Number of set bits among the low m bits of w.
def word_count(m: Nat, +w: U32) -> Nat:
  match m:
    case 0n:
      0n
    case 1n+p:
      Nat.add(bit_value(low(w)), word_count(p, U32.shr(w)))

def member_pick(b: Bool, +off: Nat, rest: List<&2, Nat>) -> List<&2, Nat>:
  match b:
    case True{}:
      Con{off, rest}
    case False{}:
      rest

# Indices (from off) of set bits among the low m bits of w, then rest.
def word_members(m: Nat, +w: U32, +off: Nat, rest: List<&2, Nat>) -> List<&2, Nat>:
  match m:
    case 0n:
      rest
    case 1n+p:
      member_pick(low(w), off, word_members(p, U32.shr(w), 1n+off, rest))

# Word-list operations.

# Zero words covering n more bits; `free` counts unused bits left in the
# last emitted word, so exactly ceil(n/32) words are produced from free = 0.
def zeros(n: Nat, free: Nat) -> List<&2, U32>:
  match n free:
    case 0n _:
      Nil{}
    case 1n+p 0n:
      Con{0, zeros(p, 31n)}
    case 1n+p 1n+f:
      zeros(p, f)

def get_pick(b: Bool, w: U32, +i: Nat, r: Bool) -> Bool:
  match b:
    case True{}:
      word_get(w, i)
    case False{}:
      r

def get_walk(ws: List<&2, U32>, +i: Nat) -> Bool:
  match ws:
    case Nil{}:
      False{}
    case Con{w, t}:
      get_pick(Nat.is_lt(i, 32n), w, i, get_walk(t, Nat.sub(i, 32n)))

def put_pick(b: Bool, +v: Bool, +w: U32, +i: Nat, t: List<&2, U32>, r: List<&2, U32>) -> List<&2, U32>:
  match b:
    case True{}:
      Con{word_put(v, w, i), t}
    case False{}:
      Con{w, r}

def put_walk(+ws: List<&2, U32>, +i: Nat, +v: Bool) -> List<&2, U32>:
  match ws:
    case Nil{}:
      Nil{}
    case Con{+w, +t}:
      put_pick(Nat.is_lt(i, 32n), v, w, i, t, put_walk(t, Nat.sub(i, 32n), v))

def zip_words(+k: WordOp, ws: List<&2, U32>, vs: List<&2, U32>) -> List<&2, U32>:
  match ws vs:
    case Con{a, s} Con{b, t}:
      Con{word_op(k, a, b), zip_words(k, s, t)}
    case _ _:
      Nil{}

def count_words(ws: List<&2, U32>) -> Nat:
  match ws:
    case Nil{}:
      0n
    case Con{w, t}:
      Nat.add(word_count(32n, w), count_words(t))

def members_words(ws: List<&2, U32>, +off: Nat) -> List<&2, Nat>:
  match ws:
    case Nil{}:
      Nil{}
    case Con{w, t}:
      word_members(32n, w, off, members_words(t, Nat.add(32n, off)))

# Public API.

# All-zero bitset of logical size n.
def new(+n: Nat) -> Bitset:
  BS{n, zeros(n, 0n)}

def length(s: Bitset) -> Nat:
  match s:
    case BS{n, ws}:
      n

def get_if(ok: Bool, ws: List<&2, U32>, i: Nat) -> Result<&2, &2, E.Error, Bool>:
  match ok:
    case True{}:
      Done{get_walk(ws, i)}
    case False{}:
      Fail{E.IndexOutOfRange{}}

def get(s: Bitset, +i: Nat) -> Result<&2, &2, E.Error, Bool>:
  match s:
    case BS{+n, ws}:
      get_if(Nat.is_lt(i, n), ws, i)

def assign_if(ok: Bool, +n: Nat, +ws: List<&2, U32>, +i: Nat, +v: Bool) -> Bitset & Result<&2, &2, E.Error, Unit>:
  match ok:
    case True{}:
      (BS{n, put_walk(ws, i, v)}, Done{Unit{}})
    case False{}:
      (BS{n, ws}, Fail{E.IndexOutOfRange{}})

def assign(s: Bitset, +i: Nat, +v: Bool) -> Bitset & Result<&2, &2, E.Error, Unit>:
  match s:
    case BS{+n, +ws}:
      assign_if(Nat.is_lt(i, n), n, ws, i, v)

# Set bit i to 1.
def set(s: Bitset, +i: Nat) -> Bitset & Result<&2, &2, E.Error, Unit>:
  assign(s, i, True{})

# Set bit i to 0.
def clear(s: Bitset, +i: Nat) -> Bitset & Result<&2, &2, E.Error, Unit>:
  assign(s, i, False{})

def count(s: Bitset) -> Nat:
  match s:
    case BS{n, ws}:
      count_words(ws)

def combine_if(ok: Bool, +k: WordOp, +n: Nat, +ws: List<&2, U32>, vs: List<&2, U32>) -> Bitset & Result<&2, &2, E.Error, Unit>:
  match ok:
    case True{}:
      (BS{n, zip_words(k, ws, vs)}, Done{Unit{}})
    case False{}:
      (BS{n, ws}, Fail{E.LengthMismatch{}})

def combine(+k: WordOp, a: Bitset, b: Bitset) -> Bitset & Result<&2, &2, E.Error, Unit>:
  match a b:
    case BS{+n, +ws} BS{+m, vs}:
      combine_if(Nat.is_eq(n, m), k, n, ws, vs)

def union(a: Bitset, b: Bitset) -> Bitset & Result<&2, &2, E.Error, Unit>:
  combine(KOr{}, a, b)

def intersection(a: Bitset, b: Bitset) -> Bitset & Result<&2, &2, E.Error, Unit>:
  combine(KAnd{}, a, b)

def difference(a: Bitset, b: Bitset) -> Bitset & Result<&2, &2, E.Error, Unit>:
  combine(KDiff{}, a, b)

def xor(a: Bitset, b: Bitset) -> Bitset & Result<&2, &2, E.Error, Unit>:
  combine(KXor{}, a, b)

# Indices of set bits, ascending.
def to_list(s: Bitset) -> List<&2, Nat>:
  match s:
    case BS{n, ws}:
      members_words(ws, 0n)

def fill_pick(b: Bool, +s: Bitset, +i: Nat) -> Bitset:
  match b:
    case True{}:
      Pair.fst(Bitset, Result<&2, &2, E.Error, Unit>, set(s, i))
    case False{}:
      s

def fill(bs: List<&2, Bool>, +i: Nat, +s: Bitset) -> Bitset:
  match bs:
    case Nil{}:
      s
    case Con{b, t}:
      fill(t, 1n+i, fill_pick(b, s, i))

def bool_count(bs: List<&2, Bool>) -> Nat:
  match bs:
    case Nil{}:
      0n
    case Con{b, t}:
      1n+bool_count(t)

# Bitset whose logical bits are exactly bs (bit 0 first).
def from_bools(+bs: List<&2, Bool>) -> Bitset:
  fill(bs, 0n, new(bool_count(bs)))

# Trace runner.

def obs_unit(r: Bitset & Result<&2, &2, E.Error, Unit>) -> Bitset & E.Obs:
  (s, x) = r
  (s, E.OUnit{x})

def step(+s: Bitset, op: E.Op) -> Bitset & E.Obs:
  match op:
    case E.Length{}:
      (s, E.ONat{length(s)})
    case E.Get{i}:
      (s, E.OBit{get(s, i)})
    case E.Set{i}:
      obs_unit(set(s, i))
    case E.Clear{i}:
      obs_unit(clear(s, i))
    case E.Count{}:
      (s, E.ONat{count(s)})
    case E.Union{ys}:
      obs_unit(union(s, from_bools(ys)))
    case E.Intersection{ys}:
      obs_unit(intersection(s, from_bools(ys)))
    case E.Difference{ys}:
      obs_unit(difference(s, from_bools(ys)))
    case E.Xor{ys}:
      obs_unit(xor(s, from_bools(ys)))
    case E.ToList{}:
      (s, E.OList{to_list(s)})

def record(acc: List<&2, E.Obs>, r: Bitset & E.Obs) -> Bitset & List<&2, E.Obs>:
  (s, o) = r
  (s, Con{o, acc})

def step_acc(op: E.Op, st: Bitset & List<&2, E.Obs>) -> Bitset & List<&2, E.Obs>:
  (s, acc) = st
  record(acc, step(s, op))

def run_acc(ops: List<&2, E.Op>, st: Bitset & List<&2, E.Obs>) -> Bitset & List<&2, E.Obs>:
  match ops:
    case Nil{}:
      st
    case Con{op, rest}:
      run_acc(rest, step_acc(op, st))

def finish(st: Bitset & List<&2, E.Obs>) -> Bitset & List<&2, E.Obs>:
  (s, acc) = st
  (s, List.reverse(&2, E.Obs, acc))

def run(ops: List<&2, E.Op>, s: Bitset) -> Bitset & List<&2, E.Obs>:
  finish(run_acc(ops, (s, Nil{})))
