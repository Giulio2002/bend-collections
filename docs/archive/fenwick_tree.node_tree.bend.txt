import Base
import ./internal/vec.bend as V
import ../types/fenwick_tree.bend as E

# Fenwick (binary indexed) tree over n U32 values; all sums wrap modulo 2^32
# (U32.add / U32.sub).
#
# Representation: the Fenwick partial sums kept in an explicit perfect tree
# of depth d (2^d >= n): every internal node stores the sum of its left
# subtree, exactly the quantity an array-based Fenwick tree stores at the
# node where a left block ends. Point addition updates the left-sums on the
# root-to-leaf path; a prefix sum adds the left-sums where the path turns
# right. Range sums are prefix differences (mod 2^32).
#
# Errors never change the state: add with index >= n fails with
# IndexOutOfRange; prefix_sum(e) with e > n and range_sum(l, r) unless
# l <= r <= n fail with InvalidRange.
#
# Cost: add / prefix_sum / range_sum O(log n); new O(log n) (untouched
# subtrees are shared); from_list O(n log n); length O(1).

type Node is Data:
  FLeaf{value: U32}
  FNode{left_sum: U32, lo: Node, hi: Node}

type Fenwick is Data:
  FT{n: Nat, depth: Nat, root: Node}

def zeros(d: Nat) -> Node:
  match d:
    case 0n:
      FLeaf{0}
    case 1n+p:
      +z = zeros(p)
      FNode{0, z, z}

# n zero values.
def new(+n: Nat) -> Fenwick:
  +d = V.depth_for(n)
  FT{n, d, zeros(d)}

def length(t: Fenwick) -> Nat:
  match t:
    case FT{n, d, r}:
      n

def add_go(d: Nat, t: Node, +i: Nat, +v: U32, left: Bool) -> Node:
  match d t left:
    case 0n FLeaf{x} _:
      FLeaf{U32.add(x, v)}
    case 0n FNode{s, l, r} _:
      FNode{s, l, r}
    case 1n+p FLeaf{x} _:
      FLeaf{x}
    case 1n+ +p FNode{s, l, r} True{}:
      FNode{U32.add(s, v), add_go(p, l, i, v, V.dec(p, i)), r}
    case 1n+ +p FNode{s, l, r} False{}:
      FNode{s, l, add_go(p, r, Nat.sub(i, V.pow2(p)), v, V.dec(p, Nat.sub(i, V.pow2(p))))}

def leaf_prefix(x: U32, i: Nat) -> U32:
  match i:
    case 0n:
      0
    case 1n+j:
      x

# Sum of the first i leaves (i <= 2^d).
def prefix_go(d: Nat, t: Node, +i: Nat, left: Bool) -> U32:
  match d t left:
    case 0n FLeaf{x} _:
      leaf_prefix(x, i)
    case 0n FNode{s, l, r} _:
      0
    case 1n+p FLeaf{x} _:
      0
    case 1n+ +p FNode{s, l, r} True{}:
      prefix_go(p, l, i, V.dec(p, i))
    case 1n+ +p FNode{s, l, r} False{}:
      U32.add(s, prefix_go(p, r, Nat.sub(i, V.pow2(p)), V.dec(p, Nat.sub(i, V.pow2(p)))))

def add_if(ok: Bool, +n: Nat, +d: Nat, +t: Node, +i: Nat, +v: U32) -> Fenwick & Result<&2, &2, E.Error, Unit>:
  match ok:
    case True{}:
      (FT{n, d, add_go(d, t, i, v, V.dec(d, i))}, Done{Unit{}})
    case False{}:
      (FT{n, d, t}, Fail{E.IndexOutOfRange{}})

# Add delta to the value at index i.
def add(f: Fenwick, +i: Nat, +v: U32) -> Fenwick & Result<&2, &2, E.Error, Unit>:
  match f:
    case FT{+n, +d, +t}:
      add_if(Nat.is_lt(i, n), n, d, t, i, v)

def prefix(+d: Nat, t: Node, +e: Nat) -> U32:
  prefix_go(d, t, e, V.dec(d, e))

def prefix_if(ok: Bool, +d: Nat, t: Node, +e: Nat) -> Result<&2, &2, E.Error, U32>:
  match ok:
    case True{}:
      Done{prefix(d, t, e)}
    case False{}:
      Fail{E.InvalidRange{}}

# Sum of the values with index in [0, e).
def prefix_sum(f: Fenwick, +e: Nat) -> Result<&2, &2, E.Error, U32>:
  match f:
    case FT{+n, +d, t}:
      prefix_if(Nat.is_le(e, n), d, t, e)

def range_if(ok: Bool, +d: Nat, +t: Node, +l: Nat, +r: Nat) -> Result<&2, &2, E.Error, U32>:
  match ok:
    case True{}:
      Done{U32.sub(prefix(d, t, r), prefix(d, t, l))}
    case False{}:
      Fail{E.InvalidRange{}}

# Sum of the values with index in [l, r).
def range_sum(f: Fenwick, +l: Nat, +r: Nat) -> Result<&2, &2, E.Error, U32>:
  match f:
    case FT{+n, +d, +t}:
      range_if(Bool.and(Nat.is_le(l, r), Nat.is_le(r, n)), d, t, l, r)

def fill(xs: List<&2, U32>, +i: Nat, +f: Fenwick) -> Fenwick:
  match xs:
    case Nil{}:
      f
    case Con{x, rest}:
      fill(rest, 1n+i, Pair.fst(Fenwick, Result<&2, &2, E.Error, Unit>, add(f, i, x)))

def count(xs: List<&2, U32>) -> Nat:
  match xs:
    case Nil{}:
      0n
    case Con{x, rest}:
      1n+count(rest)

# Fenwick tree holding exactly the given values.
def from_list(+xs: List<&2, U32>) -> Fenwick:
  fill(xs, 0n, new(count(xs)))

# Trace runner.

def obs_unit(r: Fenwick & Result<&2, &2, E.Error, Unit>) -> Fenwick & E.Obs:
  (f, x) = r
  (f, E.OUnit{x})

def step(+f: Fenwick, op: E.Op) -> Fenwick & E.Obs:
  match op:
    case E.Length{}:
      (f, E.ONat{length(f)})
    case E.Add{i, v}:
      obs_unit(add(f, i, v))
    case E.PrefixSum{e}:
      (f, E.OSum{prefix_sum(f, e)})
    case E.RangeSum{l, r}:
      (f, E.OSum{range_sum(f, l, r)})

def record(acc: List<&2, E.Obs>, r: Fenwick & E.Obs) -> Fenwick & List<&2, E.Obs>:
  (f, o) = r
  (f, Con{o, acc})

def step_acc(op: E.Op, st: Fenwick & List<&2, E.Obs>) -> Fenwick & List<&2, E.Obs>:
  (f, acc) = st
  record(acc, step(f, op))

def run_acc(ops: List<&2, E.Op>, st: Fenwick & List<&2, E.Obs>) -> Fenwick & List<&2, E.Obs>:
  match ops:
    case Nil{}:
      st
    case Con{op, rest}:
      run_acc(rest, step_acc(op, st))

def finish(st: Fenwick & List<&2, E.Obs>) -> Fenwick & List<&2, E.Obs>:
  (f, acc) = st
  (f, List.reverse(&2, E.Obs, acc))

def run(ops: List<&2, E.Op>, f: Fenwick) -> Fenwick & List<&2, E.Obs>:
  finish(run_acc(ops, (f, Nil{})))
