# Union-find: public proof entry point.
#   abstraction  ST.abs  (the representative of each element 0 .. n-1)
#   invariant    ST.inv  (perfect vector covering n; every stored
#                representative is a fixed point whose cell lists exactly its
#                class with its exact size; count = number of classes)
#   union        UN.* (relinking the smaller class = the spec relabelling)
#   operations   ST.step_ok (every op; any index >= n gives OutOfRange)
#   traces       trace_new / trace_new_inv (arbitrary op lists from U.new(n))

def new_abs(+n: Nat) -> {ST.abs(U.new(n)) == S.new(n) : List<&2, Nat>}:
  IN.new_abs(n)

def new_inv(+n: Nat) -> {ST.inv(U.new(n)) == True{} : Bool}:
  IN.new_inv(n)

def step_ok(+s: U.UnionFind, +op: E.Op, +g: {ST.inv(s) == True{} : Bool}) -> ST.StepOK(s, op):
  ST.step_ok(s, op, g)

def trace_new(+n: Nat, +ops: List<&2, E.Op>) -> {UT.view(U.run(ops, U.new(n))) == S.run(ops, S.new(n)) : List<&2, Nat> & List<&2, E.Obs>}:
  %IN.new_abs(n) : {UT.view(U.run(ops, U.new(n))) == S.run(ops, _) : List<&2, Nat> & List<&2, E.Obs>}
  UT.trace_from(ops, U.new(n), IN.new_inv(n))

def trace_new_inv(+n: Nat, +ops: List<&2, E.Op>) -> {ST.inv(Pair.fst(U.UnionFind, List<&2, E.Obs>, U.run(ops, U.new(n)))) == True{} : Bool}:
  UT.inv_from(ops, U.new(n), IN.new_inv(n))
