/* Optimized C reference for the retained LRU cache (src/lru/, the natively
 * compilable port of reference/lru/src/cache.bend).
 *
 * SAME ALGORITHM AND SAME REPRESENTATION CHOICES as the Bend side, which is
 * what "same-algorithm reference" means here:
 *
 *   - the entries live in a map keyed by the ENCODED STRING key. Bend's
 *     `Base.Map` is a binary trie over the key's bits (`MTip | MLeaf{key,val}
 *     | MNode{pos, lo, hi}`), so this reference uses a crit-bit trie with the
 *     same shape: a leaf holds the whole key, an internal node holds the bit
 *     position that distinguishes its two subtrees.
 *   - the recency order is a LIST of encoded keys, oldest first, exactly as
 *     `Cache.order` is. Touching a key removes it from that list and appends
 *     it at the back, both by walking the list. An intrusive doubly linked
 *     list with O(1) touch would be a DIFFERENT algorithm from the retained
 *     snapshot and is deliberately not used; the difference is recorded in
 *     docs/C_EQUIVALENCE.md.
 *   - the five metric counters are 64 bit and wrap, as `Metrics` does.
 *
 * Memory is a bump arena, the normal efficient choice for a short-lived C
 * benchmark. Nothing here is slowed down on purpose.
 *
 * argv = [op, size, count, reps, seed, order]; selectors are the ones
 * benchmarks/experiments/lru_workloads.py documents:
 *   0 add, 1 get, 2 peek, 3 contains, 4 remove, 5 purge, 6 resize,
 *   7 keys, 8 len, 9 new, 10 remove+add (the restoring pair), else null.
 */
#include "common.h"

#define NULL_OP 99u
#define KEYMAX 16

typedef struct Node {
  int internal;          /* 1: crit-bit node, 0: leaf */
  uint32_t pos;          /* bit position, internal only */
  struct Node *lo, *hi;  /* children, internal only */
  char key[KEYMAX];      /* leaf only */
  uint32_t val;          /* leaf only */
  int live;              /* leaf only: 0 once removed */
} Node;

typedef struct KeyCell {
  char key[KEYMAX];
  struct KeyCell *next;
} KeyCell;

typedef struct {
  uint32_t rng, chk;
  uint32_t capacity;
  uint32_t len;
  Node *root;
  KeyCell *head, *tail;   /* oldest first, like `order` */
  uint64_t inserts, evictions, removals, hits, misses;
} St;

/* ---- the encoded key: "k" followed by the decimal index, as the Bend
 * driver's `Codec.string_encode` produces for the same argument ---- */
static void key_of(char *out, uint32_t i) {
  int n = 0;
  char digits[12];
  out[n++] = 'k';
  if (i == 0) {
    out[n++] = '0';
  } else {
    int d = 0;
    while (i > 0) { digits[d++] = (char)('0' + (i % 10u)); i /= 10u; }
    while (d-- > 0) out[n++] = digits[d];
  }
  out[n] = '\0';
}

static inline int key_eq(const char *a, const char *b) {
  return strcmp(a, b) == 0;
}

/* bit `p` of the key (byte p/8, bit p%8), 0 past the end */
static inline int key_bit(const char *k, uint32_t p) {
  uint32_t byte = p >> 3;
  if (byte >= KEYMAX) return 0;
  unsigned char c = (unsigned char)k[byte];
  if (c == 0 && byte > 0) return 0;
  return (c >> (7u - (p & 7u))) & 1u;
}

static uint32_t first_diff(const char *a, const char *b) {
  for (uint32_t byte = 0; byte < KEYMAX; byte++) {
    unsigned char x = (unsigned char)a[byte], y = (unsigned char)b[byte];
    if (x != y) {
      for (uint32_t bit = 0; bit < 8; bit++) {
        uint32_t p = byte * 8u + bit;
        if (key_bit(a, p) != key_bit(b, p)) return p;
      }
    }
  }
  return 0;
}

static Node *leaf_new(const char *k, uint32_t v) {
  Node *n = (Node *)arena_alloc(sizeof(Node));
  n->internal = 0;
  memcpy(n->key, k, KEYMAX);
  n->val = v;
  n->live = 1;
  n->lo = n->hi = NULL;
  n->pos = 0;
  return n;
}

static Node *trie_find(Node *t, const char *k) {
  while (t && t->internal) t = key_bit(k, t->pos) ? t->hi : t->lo;
  if (t && t->live && key_eq(t->key, k)) return t;
  return NULL;
}

static Node *trie_insert(Node *t, const char *k, uint32_t v) {
  if (!t) return leaf_new(k, v);
  /* find the leaf this key would reach */
  Node *l = t;
  while (l->internal) l = key_bit(k, l->pos) ? l->hi : l->lo;
  if (key_eq(l->key, k)) { l->val = v; l->live = 1; return t; }
  uint32_t p = first_diff(l->key, k);
  Node *leaf = leaf_new(k, v);
  Node *node = (Node *)arena_alloc(sizeof(Node));
  node->internal = 1;
  node->pos = p;
  node->live = 1;
  node->val = 0;
  node->key[0] = '\0';
  /* re-descend, splicing the new node in at the first position > p */
  Node **slot = &t;
  while ((*slot)->internal && (*slot)->pos < p)
    slot = key_bit(k, (*slot)->pos) ? &(*slot)->hi : &(*slot)->lo;
  if (key_bit(k, p)) { node->lo = *slot; node->hi = leaf; }
  else { node->lo = leaf; node->hi = *slot; }
  *slot = node;
  return t;
}

/* removal marks the leaf dead, which is what `Map.set(.., None)` does on the
 * Bend side (the map keeps the key, the entry becomes None) */
static void trie_erase(Node *t, const char *k) {
  Node *l = trie_find(t, k);
  if (l) l->live = 0;
}

/* ---- the recency list, oldest first ---- */

static void order_remove(St *s, const char *k) {
  KeyCell *p = NULL, *c = s->head;
  while (c) {
    if (key_eq(c->key, k)) {
      if (p) p->next = c->next; else s->head = c->next;
      if (s->tail == c) s->tail = p;
      return;
    }
    p = c;
    c = c->next;
  }
}

static void order_append(St *s, const char *k) {
  KeyCell *c = (KeyCell *)arena_alloc(sizeof(KeyCell));
  memcpy(c->key, k, KEYMAX);
  c->next = NULL;
  if (s->tail) s->tail->next = c; else s->head = c;
  s->tail = c;
}

static void order_touch(St *s, const char *k) {
  order_remove(s, k);
  order_append(s, k);
}

static void lru_init(St *s, uint32_t seed, uint32_t capacity) {
  s->rng = seed;
  s->chk = 0;
  s->capacity = capacity;
  s->len = 0;
  s->root = NULL;
  s->head = s->tail = NULL;
  s->inserts = s->evictions = s->removals = s->hits = s->misses = 0;
}

static uint32_t evict_oldest(St *s) {
  if (!s->head) return 0;
  char k[KEYMAX];
  memcpy(k, s->head->key, KEYMAX);
  Node *l = trie_find(s->root, k);
  uint32_t v = l ? l->val : 0;
  trie_erase(s->root, k);
  order_remove(s, k);
  s->len--;
  s->evictions++;
  return v;
}

static void lru_add(St *s, const char *k, uint32_t v) {
  Node *l = trie_find(s->root, k);
  if (!l && s->len >= s->capacity && s->capacity > 0) evict_oldest(s);
  if (!l) s->len++;
  s->root = trie_insert(s->root, k, v);
  order_touch(s, k);
  s->inserts++;
}

static uint32_t lru_get(St *s, const char *k) {
  Node *l = trie_find(s->root, k);
  if (!l) { s->misses++; return 0; }
  order_touch(s, k);
  s->hits++;
  return l->val;
}

static uint32_t lru_peek(St *s, const char *k) {
  Node *l = trie_find(s->root, k);
  if (!l) { s->misses++; return 0; }
  s->hits++;
  return l->val;
}

static uint32_t lru_contains(St *s, const char *k) {
  return trie_find(s->root, k) ? 1u : 0u;
}

static uint32_t lru_remove(St *s, const char *k) {
  Node *l = trie_find(s->root, k);
  if (!l) return 0;
  uint32_t v = l->val;
  trie_erase(s->root, k);
  order_remove(s, k);
  s->len--;
  s->removals++;
  return v;
}

static uint32_t lru_purge(St *s) {
  uint32_t n = 0;
  while (s->head) { evict_oldest(s); n++; }
  s->inserts = s->evictions = s->removals = s->hits = s->misses = 0;
  return n;
}

static uint32_t lru_resize(St *s, uint32_t cap) {
  s->capacity = cap;
  uint32_t evicted = 0;
  while (s->len > s->capacity) { evict_oldest(s); evicted++; }
  return mix(cap, evicted);
}

/* the keys, oldest first, folded like the Bend driver folds the list */
static uint32_t fold_keys_list(St *s, uint32_t c) {
  for (KeyCell *p = s->head; p; p = p->next) {
    uint32_t idx = 0;
    for (const char *q = p->key + 1; *q; q++) idx = idx * 10u + (uint32_t)(*q - '0');
    c = mix(c, idx);
  }
  return c;
}

/* ---- the measured operations ---- */

static inline uint32_t draw(St *s, uint32_t size) {
  uint32_t r = lcg(s->rng);
  s->rng = r;
  return r % (size + 1u);
}

static inline void op_add(St *s, uint32_t size) {
  uint32_t i = draw(s, size);
  char k[KEYMAX];
  key_of(k, i);
  int present = trie_find(s->root, k) != NULL;
  lru_add(s, k, i);
  s->chk = mix(s->chk, present ? 2u : 1u);
}

static inline void op_get(St *s, uint32_t size) {
  uint32_t i = draw(s, size);
  char k[KEYMAX];
  key_of(k, i);
  s->chk = mix(s->chk, lru_get(s, k));
}

static inline void op_peek(St *s, uint32_t size) {
  uint32_t i = draw(s, size);
  char k[KEYMAX];
  key_of(k, i);
  s->chk = mix(s->chk, lru_peek(s, k));
}

static inline void op_contains(St *s, uint32_t size) {
  uint32_t i = draw(s, size);
  char k[KEYMAX];
  key_of(k, i);
  s->chk = mix(s->chk, lru_contains(s, k));
}

static inline void op_remove(St *s, uint32_t size) {
  uint32_t i = draw(s, size);
  char k[KEYMAX];
  key_of(k, i);
  s->chk = mix(s->chk, lru_remove(s, k));
}

/* the restoring pair: remove then add the same key back */
static inline void op_remove_add(St *s, uint32_t size) {
  uint32_t i = draw(s, size);
  char k[KEYMAX];
  key_of(k, i);
  uint32_t v = lru_remove(s, k);
  lru_add(s, k, i);
  s->chk = mix(s->chk, v);
}

static inline void op_purge(St *s) {
  s->rng = lcg(s->rng);
  s->chk = mix(s->chk, lru_purge(s));
}

static inline void op_resize(St *s, uint32_t size) {
  uint32_t i = draw(s, size);
  s->chk = mix(s->chk, lru_resize(s, i));
}

static inline void op_keys(St *s) {
  s->rng = lcg(s->rng);
  s->chk = fold_keys_list(s, s->chk);
}

static inline void op_len(St *s) {
  s->rng = lcg(s->rng);
  s->chk = mix(s->chk, s->len);
}

static inline void op_new(St *s) {
  s->rng = lcg(s->rng);
  s->chk = mix(s->chk, 0);
}

static inline void op_null(St *s) {
  uint32_t r = lcg(s->rng);
  s->rng = r;
  s->chk = mix(s->chk, r);
}

static uint32_t round_lru(uint32_t op, uint32_t size, uint32_t count,
                          uint32_t nulls, uint32_t seed) {
  arena_reset();
  St s;
  lru_init(&s, seed, size ? size : 1u);
  for (uint32_t i = 0; i < size; i++) {
    char k[KEYMAX];
    key_of(k, i);
    lru_add(&s, k, i);
  }
  s.chk = fold_keys_list(&s, s.chk); /* settle */
  St *sp = &s;
  keep(sp);
  switch (op) {
    case 0: for (uint32_t i = 0; i < count; i++) { keep(sp); op_add(sp, size); } break;
    case 1: for (uint32_t i = 0; i < count; i++) { keep(sp); op_get(sp, size); } break;
    case 2: for (uint32_t i = 0; i < count; i++) { keep(sp); op_peek(sp, size); } break;
    case 3: for (uint32_t i = 0; i < count; i++) { keep(sp); op_contains(sp, size); } break;
    case 4: for (uint32_t i = 0; i < count; i++) { keep(sp); op_remove(sp, size); } break;
    case 5: for (uint32_t i = 0; i < count; i++) { keep(sp); op_purge(sp); } break;
    case 6: for (uint32_t i = 0; i < count; i++) { keep(sp); op_resize(sp, size); } break;
    case 7: for (uint32_t i = 0; i < count; i++) { keep(sp); op_keys(sp); } break;
    case 8: for (uint32_t i = 0; i < count; i++) { keep(sp); op_len(sp); } break;
    case 9: for (uint32_t i = 0; i < count; i++) { keep(sp); op_new(sp); } break;
    case 10: for (uint32_t i = 0; i < count; i++) { keep(sp); op_remove_add(sp, size); } break;
    default: for (uint32_t i = 0; i < count; i++) { keep(sp); op_null(sp); } break;
  }
  for (uint32_t i = 0; i < nulls; i++) { keep(sp); op_null(sp); }
  /* size-independent drain: sixteen peeks and the length */
  s.chk = mix(s.chk, s.rng);
  for (int i = 0; i < 16; i++) op_peek(&s, size);
  return mix(s.chk, s.len);
}

int main(int argc, char **argv) {
  bench_args a = parse_args(argc, argv);
  arena_init(4096ull * 1024ull * 1024ull);
  BENCH_REGIONS(round_lru)
  return 0;
}
