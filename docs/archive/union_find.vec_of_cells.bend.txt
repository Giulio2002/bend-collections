import Base
import ./internal/vec.bend as V
import ../types/union_find.bend as E

# Disjoint-set union over the elements 0 .. n-1.
#
# Representation: a persistent vector (src/internal/vec.bend, O(log n)
# access) of cells. Every element's cell stores its class representative
# (`root`) directly; the representative's own cell additionally stores the
# class size and the list of members.
#
# Union by size: the representative of the larger class survives (ties: the
# class of the first argument) and only the members of the smaller class are
# relinked. Because every element points straight at its representative,
# paths always have length at most one: this is the Bend-appropriate
# equivalent of path compression (in a persistent structure, compressing
# during find would make find return a new state, and the amortised bound of
# path compression does not survive reuse of old versions).
#
# Errors never change the state: any index >= n yields OutOfRange.
#
# Cost: find / connected / component_size O(log n); component_count O(1);
# union O(k log n) where k is the size of the smaller class, so any sequence
# of unions relinks each element at most log2(n) times (its class at least
# doubles each time): O(n log^2 n) total. new O(n).

type Cell is Data:
  Cell{root: Nat, size: Nat, members: List<&2, Nat>}

type UnionFind is Data:
  UF{n: Nat, depth: Nat, cells: V.Vec<Cell>, count: Nat}

def croot(c: Cell) -> Nat:
  match c:
    case Cell{r, z, ms}:
      r

def csize(c: Cell) -> Nat:
  match c:
    case Cell{r, z, ms}:
      z

def cmembers(c: Cell) -> List<&2, Nat>:
  match c:
    case Cell{r, z, ms}:
      ms

def with_root(c: Cell, +r: Nat) -> Cell:
  match c:
    case Cell{q, z, ms}:
      Cell{r, z, ms}

def cell_of(m: Maybe<&2, Cell>, +x: Nat) -> Cell:
  match m:
    case Some{c}:
      c
    case None{}:
      Cell{x, 1n, Nil{}}

def cell(+d: Nat, t: V.Vec<Cell>, +x: Nat) -> Cell:
  cell_of(V.get(Cell, d, t, x), x)

def root(+d: Nat, t: V.Vec<Cell>, +x: Nat) -> Nat:
  croot(cell(d, t, x))

# Singleton cells for slots off .. off + 2^d - 1.
def singletons(d: Nat, +off: Nat) -> V.Vec<Cell>:
  match d:
    case 0n:
      V.VLeaf{Cell{off, 1n, Con{off, Nil{}}}}
    case 1n+ +p:
      V.VNode{singletons(p, off), singletons(p, Nat.add(off, V.pow2(p)))}

# Every element of 0 .. n-1 in its own class.
def new(+n: Nat) -> UnionFind:
  +d = V.depth_for(n)
  UF{n, d, singletons(d, 0n), n}

def find_if(ok: Bool, +d: Nat, t: V.Vec<Cell>, +x: Nat) -> Result<&2, &2, E.Error, Nat>:
  match ok:
    case True{}:
      Done{root(d, t, x)}
    case False{}:
      Fail{E.OutOfRange{}}

# Representative of x's class.
def find(s: UnionFind, +x: Nat) -> Result<&2, &2, E.Error, Nat>:
  match s:
    case UF{+n, +d, t, c}:
      find_if(Nat.is_lt(x, n), d, t, x)

def both(+n: Nat, +a: Nat, +b: Nat) -> Bool:
  Bool.and(Nat.is_lt(a, n), Nat.is_lt(b, n))

def connected_if(ok: Bool, +d: Nat, +t: V.Vec<Cell>, +a: Nat, +b: Nat) -> Result<&2, &2, E.Error, Bool>:
  match ok:
    case True{}:
      Done{Nat.is_eq(root(d, t, a), root(d, t, b))}
    case False{}:
      Fail{E.OutOfRange{}}

def connected(s: UnionFind, +a: Nat, +b: Nat) -> Result<&2, &2, E.Error, Bool>:
  match s:
    case UF{+n, +d, t, c}:
      connected_if(both(n, a, b), d, t, a, b)

def size_if(ok: Bool, +d: Nat, +t: V.Vec<Cell>, +x: Nat) -> Result<&2, &2, E.Error, Nat>:
  match ok:
    case True{}:
      Done{csize(cell(d, t, root(d, t, x)))}
    case False{}:
      Fail{E.OutOfRange{}}

# Number of elements in x's class.
def component_size(s: UnionFind, +x: Nat) -> Result<&2, &2, E.Error, Nat>:
  match s:
    case UF{+n, +d, t, c}:
      size_if(Nat.is_lt(x, n), d, t, x)

# Number of classes.
def component_count(s: UnionFind) -> Nat:
  match s:
    case UF{n, d, t, c}:
      c

# Point every listed member at `big`.
def relink(ms: List<&2, Nat>, +d: Nat, +t: V.Vec<Cell>, +big: Nat) -> V.Vec<Cell>:
  match ms:
    case Nil{}:
      t
    case Con{+m, rest}:
      relink(rest, d, V.set(Cell, d, t, m, with_root(cell(d, t, m), big)), big)

def merge(+d: Nat, +t: V.Vec<Cell>, +big: Nat, +small: Nat) -> V.Vec<Cell>:
  +cb = cell(d, t, big)
  +cs = cell(d, t, small)
  V.set(Cell, d, relink(cmembers(cs), d, t, big), big, Cell{big, Nat.add(csize(cb), csize(cs)), List.append(&2, Nat, cmembers(cs), cmembers(cb))})

def merge_keep(keep_a: Bool, +d: Nat, +t: V.Vec<Cell>, +ra: Nat, +rb: Nat) -> V.Vec<Cell>:
  match keep_a:
    case True{}:
      merge(d, t, ra, rb)
    case False{}:
      merge(d, t, rb, ra)

def union_roots(same: Bool, +n: Nat, +d: Nat, +t: V.Vec<Cell>, +c: Nat, +ra: Nat, +rb: Nat) -> UnionFind & Result<&2, &2, E.Error, Bool>:
  match same:
    case True{}:
      (UF{n, d, t, c}, Done{False{}})
    case False{}:
      (UF{n, d, merge_keep(Nat.is_le(csize(cell(d, t, rb)), csize(cell(d, t, ra))), d, t, ra, rb), Nat.sub(c, 1n)}, Done{True{}})

def union_if(ok: Bool, +n: Nat, +d: Nat, +t: V.Vec<Cell>, +c: Nat, +a: Nat, +b: Nat) -> UnionFind & Result<&2, &2, E.Error, Bool>:
  match ok:
    case True{}:
      union_roots(Nat.is_eq(root(d, t, a), root(d, t, b)), n, d, t, c, root(d, t, a), root(d, t, b))
    case False{}:
      (UF{n, d, t, c}, Fail{E.OutOfRange{}})

# Join the classes of a and b; Done{True} if they were distinct.
def union(s: UnionFind, +a: Nat, +b: Nat) -> UnionFind & Result<&2, &2, E.Error, Bool>:
  match s:
    case UF{+n, +d, +t, +c}:
      union_if(both(n, a, b), n, d, t, c, a, b)

# Trace runner.

def obs_bool(r: UnionFind & Result<&2, &2, E.Error, Bool>) -> UnionFind & E.Obs:
  (s, x) = r
  (s, E.OBool{x})

def step(+s: UnionFind, op: E.Op) -> UnionFind & E.Obs:
  match op:
    case E.Find{x}:
      (s, E.OItem{find(s, x)})
    case E.Union{a, b}:
      obs_bool(union(s, a, b))
    case E.Connected{a, b}:
      (s, E.OBool{connected(s, a, b)})
    case E.ComponentSize{x}:
      (s, E.OItem{component_size(s, x)})
    case E.ComponentCount{}:
      (s, E.ONat{component_count(s)})

def record(acc: List<&2, E.Obs>, r: UnionFind & E.Obs) -> UnionFind & List<&2, E.Obs>:
  (s, o) = r
  (s, Con{o, acc})

def step_acc(op: E.Op, st: UnionFind & List<&2, E.Obs>) -> UnionFind & List<&2, E.Obs>:
  (s, acc) = st
  record(acc, step(s, op))

def run_acc(ops: List<&2, E.Op>, st: UnionFind & List<&2, E.Obs>) -> UnionFind & List<&2, E.Obs>:
  match ops:
    case Nil{}:
      st
    case Con{op, rest}:
      run_acc(rest, step_acc(op, st))

def finish(st: UnionFind & List<&2, E.Obs>) -> UnionFind & List<&2, E.Obs>:
  (s, acc) = st
  (s, List.reverse(&2, E.Obs, acc))

def run(ops: List<&2, E.Op>, s: UnionFind) -> UnionFind & List<&2, E.Obs>:
  finish(run_acc(ops, (s, Nil{})))
