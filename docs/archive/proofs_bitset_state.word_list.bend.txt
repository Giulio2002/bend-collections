# Abstraction and invariant of the packed representation.
#   flat(ws)  all stored bits, word by word, bit 0 first
#   abs       the first `len` stored bits
#   inv       len <= |flat| < len + 32 (exactly ceil(len/32) words) and
#             every stored bit at position >= len is zero (masked tail)

def flat(ws: List<&2, U32>) -> List<&2, Bool>:
  match ws:
    case Nil{}:
      Nil{}
    case Con{w, t}:
      SC.append(Bool, W.ubits(w), flat(t))

def abs(s: B.Bitset) -> List<&2, Bool>:
  match s:
    case B.BS{n, ws}:
      SC.take(Bool, flat(ws), n)

def invf(+n: Nat, +xs: List<&2, Bool>) -> Bool:
  Bool.and(Nat.is_le(n, SC.length(Bool, xs)), Bool.and(Nat.is_lt(SC.length(Bool, xs), Nat.add(32n, n)), BL.allf(SC.drop(Bool, xs, n))))

def inv(s: B.Bitset) -> Bool:
  match s:
    case B.BS{n, ws}:
      invf(n, flat(ws))

def inv_le(+n: Nat, +xs: List<&2, Bool>, +g: {invf(n, xs) == True{} : Bool}) -> {Nat.is_le(n, SC.length(Bool, xs)) == True{} : Bool}:
  L.and_left(Nat.is_le(n, SC.length(Bool, xs)), Bool.and(Nat.is_lt(SC.length(Bool, xs), Nat.add(32n, n)), BL.allf(SC.drop(Bool, xs, n))), g)

def inv_hi(+n: Nat, +xs: List<&2, Bool>, +g: {invf(n, xs) == True{} : Bool}) -> {Nat.is_lt(SC.length(Bool, xs), Nat.add(32n, n)) == True{} : Bool}:
  L.and_left(Nat.is_lt(SC.length(Bool, xs), Nat.add(32n, n)), BL.allf(SC.drop(Bool, xs, n)), L.and_right(Nat.is_le(n, SC.length(Bool, xs)), Bool.and(Nat.is_lt(SC.length(Bool, xs), Nat.add(32n, n)), BL.allf(SC.drop(Bool, xs, n))), g))

def inv_tail(+n: Nat, +xs: List<&2, Bool>, +g: {invf(n, xs) == True{} : Bool}) -> {BL.allf(SC.drop(Bool, xs, n)) == True{} : Bool}:
  L.and_right(Nat.is_lt(SC.length(Bool, xs), Nat.add(32n, n)), BL.allf(SC.drop(Bool, xs, n)), L.and_right(Nat.is_le(n, SC.length(Bool, xs)), Bool.and(Nat.is_lt(SC.length(Bool, xs), Nat.add(32n, n)), BL.allf(SC.drop(Bool, xs, n))), g))

def inv_mk(+n: Nat, +xs: List<&2, Bool>, +a: {Nat.is_le(n, SC.length(Bool, xs)) == True{} : Bool}, +b: {Nat.is_lt(SC.length(Bool, xs), Nat.add(32n, n)) == True{} : Bool}, +c: {BL.allf(SC.drop(Bool, xs, n)) == True{} : Bool}) -> {invf(n, xs) == True{} : Bool}:
  L.and_intro(Nat.is_le(n, SC.length(Bool, xs)), Bool.and(Nat.is_lt(SC.length(Bool, xs), Nat.add(32n, n)), BL.allf(SC.drop(Bool, xs, n))), a, L.and_intro(Nat.is_lt(SC.length(Bool, xs), Nat.add(32n, n)), BL.allf(SC.drop(Bool, xs, n)), b, c))

# The logical length is len.
def length_abs(+n: Nat, +xs: List<&2, Bool>, +g: {invf(n, xs) == True{} : Bool}) -> {SC.length(Bool, SC.take(Bool, xs, n)) == n : Nat}:
  BL.length_take(xs, n, inv_le(n, xs, g))

# The stored bits are the logical bits followed by the zero tail.
def split(+n: Nat, +xs: List<&2, Bool>) -> {xs == SC.append(Bool, SC.take(Bool, xs, n), SC.drop(Bool, xs, n)) : List<&2, Bool>}:
  Equal.sym(List<&2, Bool>, SC.append(Bool, SC.take(Bool, xs, n), SC.drop(Bool, xs, n)), xs, BL.take_drop(xs, n))

def take32(+w: U32) -> {SC.take(Bool, W.ubits(w), 32n) == W.ubits(w) : List<&2, Bool>}:
  L.subst(Nat, k => {SC.take(Bool, W.ubits(w), k) == W.ubits(w) : List<&2, Bool>}, SC.length(Bool, W.ubits(w)), 32n, W.ulen(w), BL.take_all(W.ubits(w)))

def length_flat_cons(+w: U32, +t: List<&2, U32>) -> {SC.length(Bool, flat(Con{w, t})) == Nat.add(32n, SC.length(Bool, flat(t))) : Nat}:
  %Equal.sym(Nat, SC.length(Bool, SC.append(Bool, W.ubits(w), flat(t))), Nat.add(SC.length(Bool, W.ubits(w)), SC.length(Bool, flat(t))), LL.length_append(Bool, W.ubits(w), flat(t))) : {_ == Nat.add(32n, SC.length(Bool, flat(t))) : Nat}
  %Equal.sym(Nat, SC.length(Bool, W.ubits(w)), 32n, W.ulen(w)) : {Nat.add(_, SC.length(Bool, flat(t))) == Nat.add(32n, SC.length(Bool, flat(t))) : Nat}
  {==}

def get_case(+w: U32, +t: List<&2, U32>, +i: Nat, b: Bool, +eb: {Nat.is_lt(i, 32n) == b : Bool}, +h: {Nat.is_lt(i, Nat.add(32n, SC.length(Bool, flat(t)))) == True{} : Bool}, ih: @+h2: {Nat.is_lt(Nat.sub(i, 32n), SC.length(Bool, flat(t))) == True{} : Bool} -> {Some{B.get_walk(t, Nat.sub(i, 32n))} == SC.nth(Bool, flat(t), Nat.sub(i, 32n)) : Maybe<&2, Bool>}) -> {Some{B.get_pick(b, w, i, B.get_walk(t, Nat.sub(i, 32n)))} == SC.nth(Bool, SC.append(Bool, W.ubits(w), flat(t)), i) : Maybe<&2, Bool>}:

      Equal.trans(Maybe<&2, Bool>, Some{B.word_get(w, i)}, SC.nth(Bool, W.ubits(w), i), SC.nth(Bool, SC.append(Bool, W.ubits(w), flat(t)), i),
        W.word_get(w, i, eb),
        Equal.sym(Maybe<&2, Bool>, SC.nth(Bool, SC.append(Bool, W.ubits(w), flat(t)), i), SC.nth(Bool, W.ubits(w), i), LL.nth_append_left(Bool, W.ubits(w), flat(t), i, hl)))

      +hr = L.subst(Nat, k => {Nat.is_le(k, i) == True{} : Bool}, 32n, SC.length(Bool, W.ubits(w)), Equal.sym(Nat, SC.length(Bool, W.ubits(w)), 32n, W.ulen(w)), le)

def get_walk(+ws: List<&2, U32>, +i: Nat, +h: {Nat.is_lt(i, SC.length(Bool, flat(ws))) == True{} : Bool}) -> {Some{B.get_walk(ws, i)} == SC.nth(Bool, flat(ws), i) : Maybe<&2, Bool>}:
  match ws:
    case Nil{}:
      Empty.absurd({Some{B.get_walk(Nil{}, i)} == SC.nth(Bool, flat(Nil{}), i) : Maybe<&2, Bool>}, N.lt_zero_absurd(i, h))
    case Con{w, t}:
      get_case(w, t, i, Nat.is_lt(i, 32n), {==}, L.subst(Nat, k => {Nat.is_lt(i, k) == True{} : Bool}, SC.length(Bool, flat(Con{w, t})), Nat.add(32n, SC.length(Bool, flat(t))), length_flat_cons(w, t), h), h2 => get_walk(t, Nat.sub(i, 32n), h2))

def put_walk(+ws: List<&2, U32>, +i: Nat, +v: Bool) -> {flat(B.put_walk(ws, i, v)) == SC.update(Bool, flat(ws), i, v) : List<&2, Bool>}:

# ---- the zero word list of new(n) ----

def zeros_allf(+n: Nat, +f: Nat) -> {BL.allf(flat(B.zeros(n, f))) == True{} : Bool}:
  match n f:
    case 0n _:
      {==}
    case 1n+p 0n:
      BL.allf_append(W.ubits(0), flat(B.zeros(p, 31n)), {==}, zeros_allf(p, 31n))
    case 1n+p 1n+g:
      zeros_allf(p, g)

def zeros_lo(+n: Nat, +f: Nat) -> {Nat.is_le(n, Nat.add(f, SC.length(Bool, flat(B.zeros(n, f))))) == True{} : Bool}:
  match n f:
    case 0n _:
      N.zero_le(Nat.add(f, SC.length(Bool, flat(B.zeros(0n, f)))))
    case 1n+p 0n:
      %Equal.sym(Nat, SC.length(Bool, flat(Con{0, B.zeros(p, 31n)})), Nat.add(32n, SC.length(Bool, flat(B.zeros(p, 31n)))), length_flat_cons(0, B.zeros(p, 31n))) : {Nat.is_le(1n+p, _) == True{} : Bool}
      zeros_lo(p, 31n)
    case 1n+p 1n+g:
      zeros_lo(p, g)

def zeros_hi(+n: Nat, +f: Nat, +hf: {Nat.is_lt(f, 32n) == True{} : Bool}) -> {Nat.is_lt(Nat.add(f, SC.length(Bool, flat(B.zeros(n, f)))), Nat.add(32n, n)) == True{} : Bool}:
  match n f:
    case 0n _:
      %Equal.sym(Nat, Nat.add(f, 0n), f, N.add_zero(f)) : {Nat.is_lt(_, 32n) == True{} : Bool}
      hf
    case 1n+p 0n:
      %Equal.sym(Nat, SC.length(Bool, flat(Con{0, B.zeros(p, 31n)})), Nat.add(32n, SC.length(Bool, flat(B.zeros(p, 31n)))), length_flat_cons(0, B.zeros(p, 31n))) : {Nat.is_lt(_, Nat.add(32n, 1n+p)) == True{} : Bool}
      zeros_hi(p, 31n, {==})
    case 1n+p 1n+g:
      zeros_hi(p, g, N.lt_trans(g, 1n+g, 32n, N.lt_succ(g), hf))

def new_inv(+n: Nat) -> {inv(B.new(n)) == True{} : Bool}:
  inv_mk(n, flat(B.zeros(n, 0n)), zeros_lo(n, 0n), zeros_hi(n, 0n, {==}), BL.drop_allf(flat(B.zeros(n, 0n)), n, zeros_allf(n, 0n)))

def new_abs(+n: Nat) -> {abs(B.new(n)) == S.new(n) : List<&2, Bool>}:
  BL.take_allf(flat(B.zeros(n, 0n)), n, zeros_allf(n, 0n), zeros_lo(n, 0n))

# ---- invariant preservation facts ----

def invf_update(+n: Nat, +xs: List<&2, Bool>, +i: Nat, +v: Bool, +h: {Nat.is_lt(i, n) == True{} : Bool}, +g: {invf(n, xs) == True{} : Bool}) -> {invf(n, SC.update(Bool, xs, i, v)) == True{} : Bool}:
  +el = Equal.sym(Nat, SC.length(Bool, SC.update(Bool, xs, i, v)), SC.length(Bool, xs), LL.length_update(Bool, xs, i, v))
  inv_mk(n, SC.update(Bool, xs, i, v),
    L.subst(Nat, k => {Nat.is_le(n, k) == True{} : Bool}, SC.length(Bool, xs), SC.length(Bool, SC.update(Bool, xs, i, v)), el, inv_le(n, xs, g)),
    L.subst(Nat, k => {Nat.is_lt(k, Nat.add(32n, n)) == True{} : Bool}, SC.length(Bool, xs), SC.length(Bool, SC.update(Bool, xs, i, v)), el, inv_hi(n, xs, g)),
    L.subst(List<&2, Bool>, d => {BL.allf(d) == True{} : Bool}, SC.drop(Bool, xs, n), SC.drop(Bool, SC.update(Bool, xs, i, v), n), Equal.sym(List<&2, Bool>, SC.drop(Bool, SC.update(Bool, xs, i, v), n), SC.drop(Bool, xs, n), BL.drop_update(xs, i, v, n, h)), inv_tail(n, xs, g)))

def invf_zipk(+k: B.WordOp, +n: Nat, +xs: List<&2, Bool>, +ys: List<&2, Bool>, +gx: {invf(n, xs) == True{} : Bool}, +gy: {invf(n, ys) == True{} : Bool}) -> {invf(n, BL.zipk(k, xs, ys)) == True{} : Bool}:
  inv_mk(n, BL.zipk(k, xs, ys),
    BL.le_length_zipk(k, n, xs, ys, inv_le(n, xs, gx), inv_le(n, ys, gy)),
    N.le_lt_trans(SC.length(Bool, BL.zipk(k, xs, ys)), SC.length(Bool, xs), Nat.add(32n, n), BL.length_zipk_le(k, xs, ys), inv_hi(n, xs, gx)),
    %Equal.sym(List<&2, Bool>, SC.drop(Bool, BL.zipk(k, xs, ys), n), BL.zipk(k, SC.drop(Bool, xs, n), SC.drop(Bool, ys, n)), BL.drop_zipk(k, xs, ys, n)) : {BL.allf(_) == True{} : Bool}
    BL.allf_zipk(k, SC.drop(Bool, xs, n), SC.drop(Bool, ys, n), inv_tail(n, xs, gx), inv_tail(n, ys, gy)))

# put_walk as seen through the abstraction.
def put_abs(+n: Nat, +ws: List<&2, U32>, +i: Nat, +v: Bool) -> {SC.take(Bool, flat(B.put_walk(ws, i, v)), n) == SC.update(Bool, SC.take(Bool, flat(ws), n), i, v) : List<&2, Bool>}:
  %Equal.sym(List<&2, Bool>, flat(B.put_walk(ws, i, v)), SC.update(Bool, flat(ws), i, v), put_walk(ws, i, v)) : {SC.take(Bool, _, n) == SC.update(Bool, SC.take(Bool, flat(ws), n), i, v) : List<&2, Bool>}
  BL.take_update(flat(ws), i, v, n)

def put_inv(+n: Nat, +ws: List<&2, U32>, +i: Nat, +v: Bool, +h: {Nat.is_lt(i, n) == True{} : Bool}, +g: {invf(n, flat(ws)) == True{} : Bool}) -> {invf(n, flat(B.put_walk(ws, i, v))) == True{} : Bool}:
  L.subst(List<&2, Bool>, x => {invf(n, x) == True{} : Bool}, SC.update(Bool, flat(ws), i, v), flat(B.put_walk(ws, i, v)), Equal.sym(List<&2, Bool>, flat(B.put_walk(ws, i, v)), SC.update(Bool, flat(ws), i, v), put_walk(ws, i, v)), invf_update(n, flat(ws), i, v, h, g))
