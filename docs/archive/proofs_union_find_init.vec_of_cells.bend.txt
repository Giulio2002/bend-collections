def cell0(+y: Nat) -> U.Cell:
  U.Cell{y, 1n, Con{y, Nil{}}}

def sing_perfect(+d: Nat, +off: Nat) -> {VL.perfect(U.Cell, d, U.singletons(d, off)) == True{} : Bool}:
  match d:
    case 0n:
      {==}
    case 1n+p:
      L.and_intro(VL.perfect(U.Cell, p, U.singletons(p, off)), VL.perfect(U.Cell, p, U.singletons(p, Nat.add(off, V.pow2(p)))), sing_perfect(p, off), sing_perfect(p, Nat.add(off, V.pow2(p))))

  match b:
    case True{}:
      +el = VL.left_len(U.Cell, p, U.singletons(p, off), sing_perfect(p, off))
      Equal.trans(Maybe<&2, U.Cell>, SC.nth(U.Cell, SC.append(U.Cell, VL.slots(U.Cell, U.singletons(p, off)), VL.slots(U.Cell, U.singletons(p, Nat.add(off, V.pow2(p))))), j), SC.nth(U.Cell, VL.slots(U.Cell, U.singletons(p, off)), j), Some{cell0(Nat.add(off, j))},
        LL.nth_append_left(U.Cell, VL.slots(U.Cell, U.singletons(p, off)), VL.slots(U.Cell, U.singletons(p, Nat.add(off, V.pow2(p)))), j, L.subst(Nat, k => {Nat.is_lt(j, k) == True{} : Bool}, V.pow2(p), SC.length(U.Cell, VL.slots(U.Cell, U.singletons(p, off))), el, eb)),

      +el = VL.left_len(U.Cell, p, U.singletons(p, off), sing_perfect(p, off))
      +le = N.not_lt_le(j, V.pow2(p), eb)
      +ea = Equal.trans(Nat, Nat.add(Nat.add(off, V.pow2(p)), Nat.sub(j, V.pow2(p))), Nat.add(off, Nat.add(V.pow2(p), Nat.sub(j, V.pow2(p)))), Nat.add(off, j), N.add_assoc(off, V.pow2(p), Nat.sub(j, V.pow2(p))), Equal.cong(Nat, Nat, q => Nat.add(off, q), Nat.add(V.pow2(p), Nat.sub(j, V.pow2(p))), j, N.sub_add(j, V.pow2(p), le)))
      Equal.trans(Maybe<&2, U.Cell>, SC.nth(U.Cell, SC.append(U.Cell, VL.slots(U.Cell, U.singletons(p, off)), VL.slots(U.Cell, U.singletons(p, Nat.add(off, V.pow2(p))))), j), SC.nth(U.Cell, VL.slots(U.Cell, U.singletons(p, Nat.add(off, V.pow2(p)))), Nat.sub(j, V.pow2(p))), Some{cell0(Nat.add(off, j))},

        %ea : {SC.nth(U.Cell, VL.slots(U.Cell, U.singletons(p, Nat.add(off, V.pow2(p)))), Nat.sub(j, V.pow2(p))) == Some{cell0(_)} : Maybe<&2, U.Cell>}

def sing_nth(+d: Nat, +off: Nat, +j: Nat, +hj: {Nat.is_lt(j, V.pow2(d)) == True{} : Bool}) -> {SC.nth(U.Cell, VL.slots(U.Cell, U.singletons(d, off)), j) == Some{cell0(Nat.add(off, j))} : Maybe<&2, U.Cell>}:
  match d:
    case 0n:
      match j:
        case 0n:
          %Equal.sym(Nat, Nat.add(off, 0n), off, N.add_zero(off)) : {Some{cell0(off)} == Some{cell0(_)} : Maybe<&2, U.Cell>}
          {==}
        case 1n+q:
          Empty.absurd({SC.nth(U.Cell, Con{cell0(off), Nil{}}, 1n+q) == Some{cell0(Nat.add(off, 1n+q))} : Maybe<&2, U.Cell>}, N.lt_zero_absurd(q, hj))
    case 1n+ +p:
      sing_case(p, off, j, hj, Nat.is_lt(j, V.pow2(p)), {==}, q => sing_nth(p, off, j, q), q => sing_nth(p, Nat.add(off, V.pow2(p)), Nat.sub(j, V.pow2(p)), q))

def cl0(+d: Nat) -> List<&2, U.Cell>:
  VL.slots(U.Cell, U.singletons(d, 0n))

def lc0(+d: Nat, +y: Nat, +hy: {Nat.is_lt(y, V.pow2(d)) == True{} : Bool}) -> {M.lc(cl0(d), y) == cell0(y) : U.Cell}:
  Equal.cong(Maybe<&2, U.Cell>, U.Cell, m => U.cell_of(m, y), SC.nth(U.Cell, cl0(d), y), Some{cell0(y)}, sing_nth(d, 0n, y, hy))

def rt0(+d: Nat, +y: Nat, +hy: {Nat.is_lt(y, V.pow2(d)) == True{} : Bool}) -> {M.rt(cl0(d), y) == y : Nat}:
  Equal.cong(U.Cell, Nat, c => U.croot(c), M.lc(cl0(d), y), cell0(y), lc0(d, y, hy))
