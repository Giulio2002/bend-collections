import Base

# Specialize the native Map lookup's result to Maybe<U32>. Storage and key
# routing remain Base.Map; avoid erasing/reboxing the slot at every branch.
def lo(pos: Nat, hi: Map<&2, Maybe<&2, U32>>, r: Map<&2, Maybe<&2, U32>> & Maybe<&2, U32>) -> Map<&2, Maybe<&2, U32>> & Maybe<&2, U32>:
  (low, value) = r
  (MNode{pos, low, hi}, value)

def hi(pos: Nat, low: Map<&2, Maybe<&2, U32>>, r: Map<&2, Maybe<&2, U32>> & Maybe<&2, U32>) -> Map<&2, Maybe<&2, U32>> & Maybe<&2, U32>:
  (high, value) = r
  (MNode{pos, low, high}, value)

def leaf(+value: Maybe<&2, U32>, r: (String & String) & Cmp) -> Map<&2, Maybe<&2, U32>> & Maybe<&2, U32>:
  ((key, stored), order) = r
  match order:
    case EQ{}:
      (MLeaf{stored, value}, value)
    case _:
      (MLeaf{stored, value}, None{})

def route(low: String -> (Map<&2, Maybe<&2, U32>> & Maybe<&2, U32>), high: String -> (Map<&2, Maybe<&2, U32>> & Maybe<&2, U32>), r: String & Bool) -> Map<&2, Maybe<&2, U32>> & Maybe<&2, U32>:
  (key, direction) = r
  match direction:
    case False{}:
      low(key)
    case True{}:
      high(key)

def get(m: Map<&2, Maybe<&2, U32>>, key: String) -> Map<&2, Maybe<&2, U32>> & Maybe<&2, U32>:
  match m:
    case MTip{}:
      (MTip{}, None{})
    case MLeaf{stored, value}:
      leaf(value, String.cmp(key, stored))
    case MNode{+pos, +low, +high}:
      route(k => lo(pos, high, get(low, k)), k => hi(pos, low, get(high, k)), Map.bit(key, pos))
